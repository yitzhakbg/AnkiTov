1//! RetentionLog entity — immutable telemetry for each card review event.
2//!
3//! One row per graded review, ingested from Anki sync. Aggregations over
4//! these rows compute per-user/per-deck retention rates and drive the
5//! anomaly detection that produces `retention_exceptions` entries.
6
7use sea_orm::entity::prelude::*;
8
9#[derive(Clone, Debug, PartialEq, DeriveEntityModel)]
10#[sea_orm(table_name = "retention_logs")]
11pub struct Model {
12    /// Primary key (UUID string).
13    #[sea_orm(primary_key, auto_increment = false)]
14    pub id: String,
15    /// FK to `users.id` — the student who performed the review.
16    pub user_id: String,
17    /// Deck identifier the reviewed card belongs to.
18    pub deck_id: String,
19    /// Anki card identifier within the deck.
20    pub card_id: String,
21    /// Review grade / ease rating; `1` marks a lapse (Anki "again") and is
22    /// what retention computations treat as a failed recall.
23    pub grade: i32,
24    /// Scheduled interval after this review, in seconds (SM-2/FSRS).
25    pub interval_seconds: i64,
26    /// SM-2 ease factor after this review (Anki stores the value ×1000).
27    pub ease_factor: f64,
28    /// Unix timestamp in seconds (UTC) of the review event.
29    pub review_timestamp: i64,
30}
31

33pub enum Relation {}
