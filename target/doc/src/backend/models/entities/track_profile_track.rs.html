1//! TrackProfileTrack junction entity — maps tracks to track profiles (many-to-many).
2//!
3//! Each row links one track to one profile. `sort_order` defines the relative
4//! priority or display sequence within the profile.
5
6use sea_orm::entity::prelude::*;
7
8#[derive(Clone, Debug, PartialEq, DeriveEntityModel)]
9#[sea_orm(table_name = "track_profile_tracks")]
10pub struct Model {
11    #[sea_orm(primary_key)]
12    pub id: i32,
13    /// Foreign key to `track_profiles.id`.
14    pub track_profile_id: String,
15    /// Foreign key to `tracks.id`.
16    pub track_id: String,
17    /// Display / priority order within the profile.
18    pub sort_order: i32,
19}
20
21#[derive(Copy, Clone, Debug, EnumIter, DeriveRelation)]
22pub enum Relation {
23    #[sea_orm(
24        belongs_to = "super::track_profile::Entity",
25        from = "Column::TrackProfileId",
26        to = "super::track_profile::Column::Id"
27    )]
28    TrackProfile,
29    #[sea_orm(
30        belongs_to = "super::track::Entity",
31        from = "Column::TrackId",
32        to = "super::track::Column::Id"
33    )]
34    Track,
35}
36
37impl ActiveModelBehavior for ActiveModel {}
