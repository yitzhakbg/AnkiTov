1//! TeacherInvite entity — one invite per enrolled teacher (Prong 8).
2//!
3//! Admin enrollment creates the user row (no password yet) plus this invite.
4//! The `:invite_id` + `:token` path pair is the *credential* for the public
5//! first-login wizard (no JWT exists before activation). `invite_status`
6//! classifies an invite as revoked → used → expired, in that order, so a
7//! non-NULL `revoked_at`/`used_at` wins over the `expires_at` check.
8
9use sea_orm::entity::prelude::*;
10
11#[derive(Clone, Debug, PartialEq, DeriveEntityModel)]
12#[sea_orm(table_name = "teacher_invites")]
13pub struct Model {
14    /// Invite identifier — the public `:invite_id` path segment paired with
15    /// `invite_token`.
16    #[sea_orm(primary_key, auto_increment = false)]
17    pub id: String,
18    /// FK to `users.id` of the not-yet-activated teacher account.
19    pub user_id: i64,
20    /// School captured at enrollment; displayed by the first-login wizard.
21    pub school: String,
22    /// Optional school community/grouping captured at enrollment.
23    pub community: Option<String>,
24    /// Optional intended role (server-side default applies when `None`).
25    pub role: Option<String>,
26    /// UI locale for the teacher's first-login wizard.
27    pub locale: String,
28    /// Optional contact email captured at enrollment (the user row's email
29    /// remains the authoritative login identity).
30    pub email: Option<String>,
31    /// Secret path credential — must match `:token` in first-login URLs.
32    /// Rotated by the resend endpoint, which also resets the TTL window.
33    pub invite_token: String,
34    /// Unix timestamp in seconds (UTC); reset on resend together with
35    /// `expires_at`.
36    pub created_at: i64,
37    /// Expiry (`created_at` + 14-day TTL); invites past this are invalid
38    /// unless already revoked/used.
39    pub expires_at: i64,
40    /// Unix timestamp in seconds (UTC); set when first login activates the
41    /// account — non-`None` means the invite is consumed.
42    pub used_at: Option<i64>,
43    /// Unix timestamp in seconds (UTC); non-`None` marks the invite revoked.
44    pub revoked_at: Option<i64>,
45}
46

48pub enum Relation {}
