1//! SyncStatus entity — tracks Anki sync state per user.
2//!
3//! One row per user, updated by the sync pipeline. The management console
4//! lists these rows for connectivity views, and the ask engine reads
5//! `last_sync` / `pending_changes` for missed-practice and stale-client
6//! detection.
7
8use sea_orm::entity::prelude::*;
9
10#[derive(Clone, Debug, PartialEq, DeriveEntityModel, Eq)]
11#[sea_orm(table_name = "sync_statuses")]
12pub struct Model {
13    /// Primary key (UUID string).
14    #[sea_orm(primary_key, auto_increment = false)]
15    pub id: String,
16    /// FK to `users.id` — the user this sync state row belongs to.
17    pub user_id: String,
18    /// Classification of the sync state ("active" | "idle" | "error"; the
19    /// management console's status query filters on this string).
20    pub status: String,
21    /// Unix timestamp in seconds (UTC) of the last successful sync;
22    /// `None` means never synced (counted as missed practice by the ask
23    /// engine's missed-practice query).
24    pub last_sync: Option<i64>,
25    /// Number of local changes queued for the next push to Anki Cloud.
26    pub pending_changes: i32,
27    /// Detail of the last sync failure; `None` while syncing is healthy.
28    pub error_message: Option<String>,
29}
30
31#[derive(Copy, Clone, Debug, EnumIter, DeriveRelation)]
32pub enum Relation {}
33
34impl ActiveModelBehavior for ActiveModel {}
