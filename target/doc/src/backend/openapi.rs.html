1//! Top-level OpenAPI aggregation and live serving of the API reference.
2//!
3//! Every controller route carries a `#[utoipa::path]` annotation; this module
4//! stitches them into one spec and serves it:
5//!
6//! - `GET /api/v1/openapi.json` — machine-readable OpenAPI 3.0 spec
7//! - `GET /scalar` — interactive Scalar API reference (CDN embed)
8//!
9//! The spec is always current: it is generated from the compiled annotations
10//! at request time, so it cannot drift from the running server. Registered as
11//! public routes in `App::after_routes` (outside the management JWT gate —
12//! docs are meant to be browsable without credentials).
13
14use axum::response::IntoResponse;
15use utoipa::OpenApi;
16
17use crate::controllers;
18
19/// OpenAPI spec for the endpoints outside the Management Console: auth,
20/// locale, student self-service, and the public display wall.
21///
22/// Paths are controller-relative (`/auth/logout`, `/locale`, …) — exactly how
23/// the controllers register their routes in `app.rs`; [`ApiDoc`] nests this
24/// doc under the app-wide `/api/v1` prefix.
25#[derive(OpenApi)]
26#[openapi(
27    paths(
28        controllers::auth::logout,
29        controllers::locale::get_locale,
30        controllers::locale::get_locale_info,
31        controllers::locale::list_locales,
32        controllers::locale::set_locale,
33        controllers::student::leaderboard::my_board,
34        controllers::student::leaderboard::my_stats,
35        controllers::student::leaderboard::patch_identity,
36        controllers::display::leaderboard::display_class_board,
37    ),
38    tags(
39        (name = "Auth", description = "Registration, login, and session teardown"),
40        (name = "Locale", description = "i18n data served to the frontend"),
41        (name = "Student", description = "JWT-gated, self-scoped student endpoints"),
42        (name = "Display", description = "Public classroom display wall (token in the path is the credential)"),
43    )
44)]
45pub struct PublicOpenApi;
46
47/// The complete AnkiTov API spec: public routes plus the Management Console.
48///
49/// Both groups are nested under `/api/v1` (the annotations carry
50/// controller-relative paths like `/management/decks` or `/locale`), which is
51/// exactly how Loco mounts them in `app.rs` and how the Caddyfile proxies
52/// them in production.
53#[derive(OpenApi)]
54#[openapi(
55    nest(
56        (path = "/api/v1", api = PublicOpenApi),
57        (path = "/api/v1", api = controllers::management::ManagementOpenApi),
58    ),
59    info(
60        title = "AnkiTov API",
61        description = "Zero-intervention knowledge retention infrastructure — management console, IMP capsule pipeline, telemetry, and student surfaces.",
62        contact(name = "AnkiTov", url = "https://ankitov.com"),
63        license(name = "AGPL-3.0-only"),
64    )
65)]
66pub struct ApiDoc;
67
68/// `GET /api/v1/openapi.json` — the live OpenAPI 3.0 spec.
69pub async fn openapi_json() -> impl IntoResponse {
70    axum::Json(ApiDoc::openapi())
71}
72
73/// `GET /scalar` — interactive API reference (Scalar, loaded from CDN).
74pub async fn scalar_ui() -> impl IntoResponse {
75    axum::response::Html(
76        r#"<!doctype html>
77<html>
78  <head>
79    <meta charset="utf-8" />
80    <meta name="viewport" content="width=device-width, initial-scale=1" />
81    <title>AnkiTov API Reference</title>
82  </head>
83  <body>
84    <!-- Scalar single-script embed (https://scalar.com) pointing at the live spec -->
85    <script id="api-reference" data-url="/api/v1/openapi.json"></script>
86    <script src="https://cdn.jsdelivr.net/npm/@scalar/api-reference"></script>
87  </body>
88</html>"#,
89    )
90}

92#[cfg(test)]
93mod tests {
94    use super::*;
95
96    /// The served spec must expose the real URL shapes: management under
97    /// `/api/v1/management`, public routes under `/api/v1` — matching both
98    /// the Loco route table and the Caddyfile rewrite.
99    #[test]
100    fn spec_paths_match_mounted_routes() {
101        let doc = ApiDoc::openapi();
102        let json = serde_json::to_value(&doc).expect("spec serializes");
103        let paths = json["paths"].as_object().expect("paths object");
104
105        for expected in [
106            "/api/v1/management/decks",
107            "/api/v1/management/compliance/weekly",
108            "/api/v1/auth/logout",
109            "/api/v1/locale",
110            "/api/v1/locale/set",
111            "/api/v1/student/leaderboard",
112            "/api/v1/display/class/{token}",
113        ] {
114            assert!(
115                paths.contains_key(expected),
116                "spec missing `{expected}` (spec has {} paths total)",
117                paths.len()
118            );
119        }
120    }
121}
