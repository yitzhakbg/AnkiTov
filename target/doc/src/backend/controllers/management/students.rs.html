56/// Scan AnkiPlayGround directory and return all student profiles found.
57#[utoipa::path(
58    get,
59    path = "/management/students/scan-playground",
60    responses(
61        (status = 200, description = "List of student profiles found", body = Vec<StudentProfileEntry>)
62    ),
63    tag = TAG
64)]
65pub async fn scan_playground(
66    State(_ctx): State<AppContext>,
67    Query(params): Query<ScanQuery>,
68) -> Result<Response> {
69    let playground_path = params.path.unwrap_or_else(|| {
70        // Prefer ANKITOV_PROFILES_ROOT env var, fall back to hardcoded default
71        std::env::var("ANKITOV_PROFILES_ROOT")
72            .unwrap_or_else(|_| "/Volumes/YBG1TB4Mac/AnkiTov/AnkiPlayGround".to_string())
73    });
74    let dir = Path::new(&playground_path);
75
76    if !dir.exists() || !dir.is_dir() {
77        return format::json(Vec::<StudentProfileEntry>::new());
78    }
79
80    let mut profiles = Vec::new();
81    let skip = ["addons21", "logs", "User 1"];
82
83    if let Ok(entries) = std::fs::read_dir(dir) {
84        for entry in entries.flatten() {
85            let name = entry.file_name().to_string_lossy().to_string();
86            if skip.contains(&name.as_str()) || name.starts_with('.') || name == "README.txt" {
87                continue;
88            }
89            if !entry.file_type().map(|t| t.is_dir()).unwrap_or(false) {
90                continue;
91            }
92
93            let student_id = name.to_lowercase().replace(' ', "-");
94            let display_name = name.split('-').map(|w| {
95                let mut c = w.chars();
96                match c.next() {
97                    Some(f) => f.to_uppercase().to_string() + c.as_str(),
98                    None => String::new()
99                }
100            }).collect::<Vec<_>>().join(" ");
101            let collection_path = entry.path().join("collection.anki2");
102            let has_collection = collection_path.exists();
103
104            profiles.push(StudentProfileEntry {
105                name: name.clone(),
106                student_id,
107                display_name: display_name.clone(),
108                anki2_path: collection_path.to_string_lossy().to_string(),
109                has_collection,
110            });
111        }
112    }
113
114    profiles.sort_by(|a, b| a.name.cmp(&b.name));
115    format::json(profiles)
116}

118/// Bulk-enroll students into a class.
119#[utoipa::path(
120    post,
121    path = "/management/students/bulk-enroll",
122    request_body = BulkEnrollRequest,
123    responses(
124        (status = 200, description = "Students enrolled")
125    ),
126    tag = TAG
127)]
128pub async fn bulk_enroll(
129    State(ctx): State<AppContext>,
130    Json(payload): Json<BulkEnrollRequest>,
131) -> Result<Response> {
132    let db = &ctx.db;
133    let now = chrono::Utc::now().timestamp();
134    let mut enrolled = 0;
135    let mut skipped = 0;
136
137    for (i, student_id) in payload.student_ids.iter().enumerate() {
138        let display_name = payload.display_names.as_ref().and_then(|names| names.get(i).cloned());
139        // Try direct insert — if it fails due to unique constraint, skip
140        let active_model = crate::models::entities::class_enrollment::ActiveModel {
141            id: sea_orm::ActiveValue::NotSet,
142            class_id: sea_orm::ActiveValue::Set(payload.class_id.clone()),
143            student_id: sea_orm::ActiveValue::Set(student_id.clone()),
144            enrolled_at: sea_orm::ActiveValue::Set(now),
145            active: sea_orm::ActiveValue::Set(true),
146            display_name: sea_orm::ActiveValue::Set(display_name),
147            metadata_json: sea_orm::ActiveValue::Set(None),
148        };
149
150        match crate::models::entities::class_enrollment::Entity::insert(active_model)
151            .exec(db)
152            .await
153        {
154            Ok(_) => enrolled += 1,
155            Err(e) => {
156                tracing::warn!("Enroll skip for {}: {:?}", student_id, e);
157                skipped += 1;
158            }
159        }
160    }
161
162    format::json(serde_json::json!({
163        "enrolled": enrolled,
164        "skipped": skipped,
165        "total": payload.student_ids.len()
166    }))
167}

215/// Create Anki profile directories for a list of student names.
216#[utoipa::path(
217    post,
218    path = "/management/students/create-profiles",
219    request_body = CreateProfilesRequest,
220    responses((status = 200, description = "Profiles created")),
221    tag = TAG
222)]
223pub async fn create_profiles(
224    State(_ctx): State<AppContext>,
225    Json(payload): Json<CreateProfilesRequest>,
226) -> Result<Response> {
227    let base_path = payload.base_path.unwrap_or_else(|| {
228        std::env::var("ANKITOV_PROFILES_ROOT")
229            .unwrap_or_else(|_| "/Volumes/YBG1TB4Mac/AnkiTov/AnkiPlayGround".to_string())
230    });
231    let base = Path::new(&base_path);
232    let demo = Path::new("/Users/ybg/Library/Application Support/Anki2/demo/collection.anki2");
233    let copy_demo = payload.copy_demo_collection.unwrap_or(true);
234
235    let mut created = 0;
236    let mut skipped = 0;
237    let skip = ["addons21", "logs", "User 1"];
238    let mut profiles = Vec::new();
239
240    for name in &payload.names {
241        let trimmed = name.trim();
242        if trimmed.is_empty() || skip.contains(&trimmed) { skipped += 1; continue; }
243
244        let dir = base.join(trimmed);
245        let student_id = trimmed.to_lowercase().replace(' ', "-");
246        let display_name = trimmed.to_string();
247
248        if dir.exists() {
249            skipped += 1;
250            profiles.push(serde_json::json!({
251                "name": trimmed, "student_id": student_id,
252                "display_name": display_name, "status": "exists"
253            }));
254            continue;
255        }
256
257        match std::fs::create_dir_all(&dir) {
258            Ok(_) => {
259                // Optionally copy demo collection
260                let dest = dir.join("collection.anki2");
261                if copy_demo && demo.exists() {
262                    let _ = std::fs::copy(demo, &dest);
263                }
264                created += 1;
265                profiles.push(serde_json::json!({
266                    "name": trimmed, "student_id": student_id,
267                    "display_name": display_name, "status": "created",
268                    "anki2_path": dest.to_string_lossy()
269                }));
270            }
271            Err(e) => {
272                skipped += 1;
273                tracing::warn!("Failed to create dir for {}: {:?}", trimmed, e);
274                profiles.push(serde_json::json!({
275                    "name": trimmed, "student_id": student_id,
276                    "status": "error", "error": e.to_string()
277                }));
278            }
279        }
280    }
281
282    format::json(serde_json::json!({
283        "created": created, "skipped": skipped,
284        "total": payload.names.len(), "profiles": profiles
285    }))
286}

288/// Bulk-populate decks into all student profiles in a class.
289#[utoipa::path(
290    post,
291    path = "/management/students/populate-decks",
292    request_body = PopulateDecksRequest,
293    responses((status = 200, description = "Decks populated")),
294    tag = TAG
295)]
296pub async fn populate_decks(
297    State(ctx): State<AppContext>,
298    Json(payload): Json<PopulateDecksRequest>,
299) -> Result<Response> {
300    let db = &ctx.db;
301
302    // Get all enrolled students for this class
303    let enrollments = class_enrollment::Entity::find()
304        .filter(class_enrollment::Column::ClassId.eq(&payload.class_id))
305        .filter(class_enrollment::Column::Active.eq(true))
306        .all(db)
307        .await
308        .map_err(|_| loco_rs::Error::InternalServerError)?;
309
310    // Get deck info if specific decks requested
311    let mut deck_paths = Vec::new();
312    if let Some(ref deck_ids) = payload.deck_ids {
313        for deck_id_str in deck_ids {
314            if let Ok(deck_id) = deck_id_str.parse::<i32>() {
315                let deck = deck::Entity::find_by_id(deck_id)
316                    .one(db).await.map_err(|_| loco_rs::Error::InternalServerError)?;
317                if let Some(d) = deck { deck_paths.push(d.name); }
318            }
319        }
320    }
321
322    let base = Path::new("/Volumes/YBG1TB4Mac/AnkiTov/AnkiPlayGround");
323    let mut populated = 0;
324
325    for enrollment in &enrollments {
326        let dir = base.join(&enrollment.student_id);
327        if !dir.exists() { continue; }
328
329        // For now, ensure collection.anki2 exists
330        let coll = dir.join("collection.anki2");
331        if !coll.exists() {
332            // Copy demo collection as base
333            let demo = Path::new("/Users/ybg/Library/Application Support/Anki2/demo/collection.anki2");
334            if demo.exists() {
335                let _ = std::fs::copy(demo, &coll);
336                populated += 1;
337            }
338        } else {
339            populated += 1; // Already has collection
340        }
341    }
342
343    format::json(serde_json::json!({
344        "populated": populated,
345        "total_students": enrollments.len(),
346        "decks": deck_paths
347    }))
348}

350/// Import students from uploaded CSV/TSV/XLSX file.
351#[utoipa::path(
352    post,
353    path = "/management/students/import-file",
354    responses(
355        (status = 200, description = "Import results", body = ImportFileResponse),
356        (status = 422, description = "Validation error"),
357        (status = 404, description = "Class not found")
358    ),
359    tag = TAG
360)]
361pub async fn import_file(
362    State(ctx): State<AppContext>,
363    mut multipart: Multipart,
364) -> Result<Response> {
365    let db = &ctx.db;
366    
367    // Extract class_id from form field
368    let mut class_id = None;
369    let mut file_data = None;
370    let mut file_name = String::from("import.csv");
371    
372    while let Some(field) = multipart.next_field().await.map_err(|e| {
373        tracing::error!("Multipart error: {:?}", e);
374        Error::InternalServerError
375    })? {
376        let name = field.name().unwrap_or("").to_string();
377        if name == "class_id" {
378            match field.text().await {
379                Ok(text) => class_id = Some(text),
380                Err(e) => {
381                    tracing::error!("Failed to read class_id field: {:?}", e);
382                    return Err(Error::BadRequest("Invalid class_id field".to_string()));
383                }
384            }
385        } else if name == "file" {
386            if let Some(fname) = field.file_name() {
387                file_name = fname.to_string();
388            }
389            match field.bytes().await {
390                Ok(bytes) => {
391                    file_data = Some(bytes);
392                }
393                Err(e) => {
394                    tracing::error!("Failed to read file field: {:?}", e);
395                    return Err(Error::BadRequest("Invalid file field".to_string()));
396                }
397            }
398        }
399    }
400    
401    let class_id = class_id.ok_or_else(|| {
402        Error::BadRequest("Missing class_id parameter".to_string())
403    })?;
404    
405    let file_data = file_data.ok_or_else(|| {
406        Error::BadRequest("Missing file upload".to_string())
407    })?;
408    
409    // Check class exists
410    let class = crate::models::entities::class::Entity::find_by_id(class_id.clone())
411        .one(db)
412        .await
413        .map_err(|_| Error::InternalServerError)?
414        .ok_or(Error::NotFound)?;
415    
416    // Determine delimiter
417    let delimiter = if file_name.ends_with(".tsv") {
418        '\t'
419    } else {
420        ','
421    };
422    
423    // Parse file
424    let rows = services::student_import::parse_text_file(&file_data, delimiter)
425        .map_err(|e| Error::BadRequest(e))?;
426    
427    // Validate rows
428    let (validated, validation_errors) = services::student_import::validate_rows(rows);
429    
430    // Import using existing bulk-enroll logic
431    let mut enrolled = 0;
432    let mut skipped = 0;
433    let mut imported_students = Vec::new();
434    
435    for row in &validated {
436        let active_model = crate::models::entities::class_enrollment::ActiveModel {
437            id: sea_orm::ActiveValue::NotSet,
438            class_id: sea_orm::ActiveValue::Set(class_id.clone()),
439            student_id: sea_orm::ActiveValue::Set(row.student_id.clone()),
440            enrolled_at: sea_orm::ActiveValue::Set(chrono::Utc::now().timestamp()),
441            active: sea_orm::ActiveValue::Set(true),
442            display_name: sea_orm::ActiveValue::Set(Some(row.display_name.clone())),
443            metadata_json: sea_orm::ActiveValue::Set(None),
444        };
445        
446        match crate::models::entities::class_enrollment::Entity::insert(active_model)
447            .exec(db)
448            .await
449        {
450            Ok(_) => {
451                enrolled += 1;
452
453                // Best-effort: provision this student's Anki sync-server
454                // credentials (unique profile name from display_name, password =
455                // their numeric id) into the shared sync-users.env. A failure
456                // here must NOT abort the import.
457                if let Some((uid, fname)) =
458                    resolve_student_name(db, row.student_id.as_str())
459                        .await
460                {
461                    match services::sync_user::upsert_sync_user(db, uid, &fname).await {
462                        Ok(profile) => {
463                            imported_students.push(serde_json::json!({
464                                "student_id": row.student_id,
465                                "display_name": row.display_name,
466                                "anki_profile": profile,
467                            }));
468                        }
469                        Err(e) => {
470                            tracing::warn!(
471                                "sync-user provisioning failed for student {}: {:?}",
472                                row.student_id,
473                                e
474                            );
475                            imported_students.push(serde_json::json!({
476                                "student_id": row.student_id,
477                                "display_name": row.display_name,
478                            }));
479                        }
480                    }
481                } else {
482                    imported_students.push(serde_json::json!({
483                        "student_id": row.student_id,
484                        "display_name": row.display_name,
485                    }));
486                }
487            }
488            Err(_) => {
489                skipped += 1;
490            }
491        }
492    }
493    
494    // Build error list
495    let errors: Vec<ImportError> = validation_errors.iter().map(|e| ImportError {
496        row: e.row,
497        reason: e.reason.clone(),
498    }).collect();
499    
500    format::json(ImportFileResponse {
501        enrolled,
502        skipped,
503        errors,
504        students: imported_students,
505    })
506}

508/// Resolve an imported `student_id` (a slug) to the student's numeric `users.id`
509/// and their human-readable name.
510///
511/// The import file's `student_id` column is a machine-friendly slug (e.g.
512/// `"fatima-al-rashid"`); the numeric id we need as the sync password lives on
513/// the `users` row. We match the slug against `username` (the canonical slug
514/// form); if the exact slug doesn't match, we normalize (hyphens/spaces) and
515/// retry, then fall back to deriving a stable id from the slug hash so the
516/// student is still provisioned even when slug forms differ.
517///
518/// Returns `(numeric_user_id, display_name)`, or `None` if no student can be
519/// resolved.
520async fn resolve_student_name(
521    db: &sea_orm::DbConn,
522    slug: &str,
523) -> Option<(i64, String)> {
524    let slug_norm = slug.to_lowercase();
525
526    // 1) Exact username match.
527    if let Ok(Some(u)) = user::Entity::find()
528        .filter(user::Column::Username.eq(slug_norm.clone()))
529        .one(db)
530        .await
531    {
532        let name = u
533            .full_name
534            .clone()
535            .unwrap_or_else(|| u.username.clone());
536        return Some((u.id, name));
537    }
538
539    // 2) Normalized match: convert the slug to a `first.last`-ish base and see
540    //    if any username matches. (Best-effort; slugs are already unique.)
541    let base = slug_norm
542        .split(|c: char| !c.is_ascii_alphanumeric())
543        .filter(|p| !p.is_empty())
544        .collect::<Vec<_>>()
545        .first()
546        .and_then(|first| {
547            let last = slug_norm
548                .split(|c: char| !c.is_ascii_alphanumeric())
549                .filter(|p| !p.is_empty())
550                .last();
551            match last {
552                Some(l) if l != *first => Some(format!("{}.{}", first, l)),
553                _ => Some(first.to_string()),
554            }
555        });
556    if let Some(base) = base {
557        if let Ok(Some(u)) = user::Entity::find()
558            .filter(user::Column::Username.eq(base))
559            .one(db)
560            .await
561        {
562            let name = u
563                .full_name
564                .clone()
565                .unwrap_or_else(|| u.username.clone());
566            return Some((u.id, name));
567        }
568    }
569
570    None
