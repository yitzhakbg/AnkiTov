2//!
3//! Combines Zone 1 (ForensicReader), Zone 2 (AnkiConnect), and Zone 3

5//! collection overview, per-deck breakdown, and retention exceptions.

26fn anki_client() -> AnkiConnectClient {
27    let port: u16 = std::env::var("ANKICONNECT_PORT")
28        .unwrap_or_else(|_| "8765".into())
29        .parse()
30        .unwrap_or(8765);
31    let api_key = std::env::var("ANKICONNECT_API_KEY").ok();
32    AnkiConnectClient::new(port, api_key)
33}

35fn collection_path() -> Option<String> {
36    std::env::var("ANKICOLLECTION_PATH").ok()
37}

39/// Full statistics overview for the dashboard Statistics tab.
40///
41/// Combines live AnkiConnect data with forensic read-only SQLite analysis.
42#[utoipa::path(
43    get,
44    path = "/management/stats",
45    responses(
46        (status = 200, description = "Statistics overview", body = ClassOverviewStats)
47    ),
48    tag = TAG
49)]
50pub async fn stats(_ctx: State<AppContext>) -> Result<Response> {
51    let client = anki_client();
52
53    // ── Zone 2: live per-deck counts ──────────────────────────────
54    let decks = client.deck_names().await.unwrap_or_default();
55    let mut total_cards: u32 = 0;
56    let mut total_due: u32 = 0;
57    let mut deck_breakdown: Vec<serde_json::Value> = Vec::new();
58
59    for deck_name in &decks {
60        if let Ok(stats) = client.get_deck_stats(vec![deck_name.clone()]).await {
61            let (total, due) = stats
62                .as_object()
63                .and_then(|o| o.values().find(|v| {
64                    v.get("name").and_then(|n| n.as_str()) == Some(deck_name.as_str())
65                }))
66                .map(|ds| {
67                    let t = ds.get("total_in_deck").and_then(|v| v.as_u64()).unwrap_or(0) as u32;
68                    // Due = new + learn + review
69                    let new = ds.get("new_count").and_then(|v| v.as_u64()).unwrap_or(0) as u32;
70                    let learn = ds.get("learn_count").and_then(|v| v.as_u64()).unwrap_or(0) as u32;
71                    let review = ds.get("review_count").and_then(|v| v.as_u64()).unwrap_or(0) as u32;
72                    (t, new + learn + review)
73                })
74                .unwrap_or((0, 0));
75
76            total_cards += total;
77            total_due += due;
78
79            let health_score = if total > 0 {
80                (100.0 * (1.0 - due as f64 / total as f64 * 0.5)) as u8
81            } else {
82                100
83            };
84
85            deck_breakdown.push(serde_json::json!({
86                "name": deck_name,
87                "total": total,
88                "due": due,
89                "health_score": health_score,
90                "status": if due > 10 { "warn" } else if due > 20 { "danger" } else { "ok" }
91            }));
92        }
93    }
94
95    // ── Zone 1: forensic collection metrics ───────────────────────
96    let (review_count, retention_trend) = if let Some(path) = collection_path() {
97        if let Ok(reader) = ForensicReader::open(&path) {
98            let review_count = reader
99                .cold_health_check()
100                .map(|h| h.review_count)
101                .unwrap_or(0);
102
103            // Get retention curve across all decks (use first deck as proxy).
104            // If the live AnkiConnect deck name isn't present in the forensic
105            // collection (e.g. a staged demo), fall back to the collection's
106            // own first deck so the trend still populates.
107            let mut trend: Vec<DailyRetention> = Vec::new();
108            if let Some(first) = decks.first() {
109                match reader.retention_curve(first, 14) {
110                    Ok(pts) => trend = pts
111                        .into_iter()
112                        .map(|p| DailyRetention {
113                            date: p.date,
114                            retention_rate: p.retention_rate,
115                            review_count: p.review_count,
116                        })
117                        .collect(),
118                    Err(_) => {
119                        if let Some(fb) = reader.first_deck_name() {
120                            if fb.as_str() != first.as_str() {
121                                trend = reader
122                                    .retention_curve(&fb, 14)
123                                    .unwrap_or_default()
124                                    .into_iter()
125                                    .map(|p| DailyRetention {
126                                        date: p.date,
127                                        retention_rate: p.retention_rate,
128                                        review_count: p.review_count,
129                                    })
130                                    .collect();
131                            }
132                        }
133                    }
134                }
135            } else if let Some(fb) = reader.first_deck_name() {
136                trend = reader
137                    .retention_curve(&fb, 14)
138                    .unwrap_or_default()
139                    .into_iter()
140                    .map(|p| DailyRetention {
141                        date: p.date,
142                        retention_rate: p.retention_rate,
143                        review_count: p.review_count,
144                    })
145                    .collect();
146            }
147
148            (review_count, trend)
149        } else {
150            (0, vec![])
151        }
152    } else {
153        (0, vec![])
154    };
155
156    // ── Retention exceptions (from relational DB) ────────────────
157    let exceptions_query = crate::models::entities::retention_exception::Entity::find()
158        .filter(crate::models::entities::retention_exception::Column::ResolvedAt.is_null())
159        .order_by_desc(crate::models::entities::retention_exception::Column::DetectedAt)
160        .all(&_ctx.db)
161        .await
162        .unwrap_or_default();
163
164    let exceptions: Vec<RetentionException> = exceptions_query
165        .into_iter()
166        .map(|e| RetentionException {
167            user_id: e.user_id,
168            deck_id: e.deck_id,
169            exception_type: e.exception_type,
170            retention_rate: e.retention_rate,
171            threshold: e.threshold,
172            detected_at: e.detected_at,
173        })
174        .collect();
175
176    // Compute average retention from the trend
177    let avg_retention = if retention_trend.is_empty() {
178        0.0
179    } else {
180        retention_trend.iter().map(|p| p.retention_rate).sum::<f64>()
181            / retention_trend.len() as f64
182    };
183
184    // Active users = student profiles with a collection.anki2
185    let active_users = if let Ok(dir) = std::env::var("ANKIPLAYGROUND_PATH") {
186        std::fs::read_dir(&dir)
187            .map(|entries| {
188                entries
189                    .flatten()
190                    .filter(|e| {
191                        e.file_type().map(|t| t.is_dir()).unwrap_or(false)
192                            && e.path().join("collection.anki2").exists()
193                    })
194                    .count() as u32
195            })
196            .unwrap_or(0)
197    } else {
198        0
199    };
200    format::json(serde_json::json!({
201        "overview": {
202            "total_decks": decks.len() as u32,
203            "total_cards": total_cards,
204            "total_reviews": review_count,
205            "average_retention": (avg_retention * 1000.0).round() / 1000.0,
206            "active_users": active_users,
207            "retention_trend": retention_trend,
208        },
209        "decks": deck_breakdown,
210        "exceptions": exceptions,
211    }))
212}

214/// Recent retention exceptions (from relational telemetry).
215#[utoipa::path(
216    get,
217    path = "/management/stats/exceptions",
218    responses(
219        (status = 200, description = "List of retention exceptions")
220    ),
221    tag = TAG
222)]
223pub async fn exceptions(State(ctx): State<AppContext>) -> Result<Response> {
224    // Query retention_exceptions via SeaORM
225    let rows = crate::models::entities::retention_exception::Entity::find()
226        .filter(
227            crate::models::entities::retention_exception::Column::ResolvedAt
228                .is_null(),
229        )
230        .order_by_desc(
231            crate::models::entities::retention_exception::Column::DetectedAt,
232        )
233        .limit(50)
234        .all(&ctx.db)
235        .await
236        .map_err(|e| {
237            tracing::error!("Failed to query retention exceptions: {e}");
238            loco_rs::Error::InternalServerError
239        })?;
240
241    let exceptions: Vec<RetentionException> = rows
242        .into_iter()
243        .map(|e| RetentionException {
244            user_id: e.user_id,
245            deck_id: e.deck_id,
246            exception_type: e.exception_type,
247            retention_rate: e.retention_rate,
248            threshold: e.threshold,
249            detected_at: e.detected_at,
250        })
251        .collect();
252
253    format::json(serde_json::json!({ "exceptions": exceptions }))
254}
