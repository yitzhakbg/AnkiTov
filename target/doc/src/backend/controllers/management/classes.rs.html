18/// Characters used for class codes (no ambiguous 0/O, 1/I/L).
19const CODE_CHARS: &[u8] = b"ABCDEFGHJKMNPQRSTUVWXYZ23456789";
20const CODE_LENGTH: usize = 6;

22const TAG: &str = "Management Console";

42/// Generate a random 6-character class code.
43fn generate_class_code() -> String {
44    use rand::Rng;
45    let mut rng = rand::thread_rng();
46    (0..CODE_LENGTH)
47        .map(|_| {
48            let idx = rng.gen_range(0..CODE_CHARS.len());
49            CODE_CHARS[idx] as char
50        })
51        .collect()
52}

101/// List all classes with student counts.
102#[utoipa::path(
103    get,
104    path = "/management/classes",
105    responses(
106        (status = 200, description = "List of classes", body = Vec<ClassSummary>)
107    ),
108    tag = TAG
109)]
110pub async fn list(
111    State(ctx): State<AppContext>,
112    Query(params): Query<ClassQuery>
113) -> Result<Response> {
114    let db = &ctx.db;
115
116    let mut find = class::Entity::find();
117
118    // Filter by search query if provided (LIKE match on name)
119    if let Some(q) = &params.q {
120        if !q.trim().is_empty() {
121            find = find.filter(class::Column::Name.contains(q.trim()));
122        }
123    }
124
125    let classes = find.all(db).await.map_err(|e| {
126        tracing::error!("Failed to list classes: {:?}", e);
127        loco_rs::Error::InternalServerError
128    })?;
129
130    let mut summaries = Vec::new();
131    for c in classes {
132        let student_count = class_enrollment::Entity::find()
133            .filter(class_enrollment::Column::ClassId.eq(&c.id))
134            .filter(class_enrollment::Column::Active.eq(true))
135            .count(db)
136            .await
137            .unwrap_or(0);
138
139        summaries.push(ClassSummary {
140            id: c.id,
141            name: c.name,
142            class_code: c.class_code,
143            subject_area: c.subject_area,
144            teacher_id: c.teacher_id,
145            period: c.period,
146            academic_year: c.academic_year,
147            student_count,
148            created_at: c.created_at,
149        });
150    }
151
152    format::json(summaries)
153}

155/// Create a new class.
156#[utoipa::path(
157    post,
158    path = "/management/classes",
159    request_body = CreateClassRequest,
160    responses(
161        (status = 201, description = "Class created", body = ClassSummary)
162    ),
163    tag = TAG
164)]
165pub async fn create(
166    Extension(user): Extension<AuthUser>,
167    State(ctx): State<AppContext>,
168    Json(payload): Json<CreateClassRequest>)
169     -> Result<Response> {
170    let _ = ensure_role(&user, &["teacher", "admin"])?;
171    let db = &ctx.db;
172    let now = chrono::Utc::now().timestamp();
173
174    let id = uuid::Uuid::new_v4().to_string();
175    let class_code = generate_class_code();
176
177    // Clone fields before moving into ActiveModel
178    let name = payload.name.clone();
179    let teacher_id = payload.teacher_id.clone();
180    let subject_area = payload.subject_area.clone();
181    let period = payload.period.clone();
182    let academic_year = payload.academic_year.clone();
183
184    let active_model = class::ActiveModel {
185        id: sea_orm::ActiveValue::Set(id.clone()),
186        name: sea_orm::ActiveValue::Set(payload.name),
187        class_code: sea_orm::ActiveValue::Set(Some(class_code.clone())),
188        subject_area: sea_orm::ActiveValue::Set(payload.subject_area),
189        teacher_id: sea_orm::ActiveValue::Set(payload.teacher_id),
190        period: sea_orm::ActiveValue::Set(payload.period),
191        academic_year: sea_orm::ActiveValue::Set(payload.academic_year),
192        created_at: sea_orm::ActiveValue::Set(now),
193        updated_at: sea_orm::ActiveValue::Set(now),
194    };
195
196    class::Entity::insert(active_model)
197        .exec(db)
198        .await
199        .map_err(|e| {
200            tracing::error!("Failed to create class: {:?}", e);
201            loco_rs::Error::InternalServerError
202        })?;
203
204    let summary = ClassSummary {
205        id,
206        name,
207        class_code: Some(class_code),
208        subject_area,
209        teacher_id,
210        period,
211        academic_year,
212        student_count: 0,
213        created_at: now,
214    };
215
216    format::json(summary)
217}

219/// Update class metadata.
220#[utoipa::path(
221    put,
222    path = "/management/classes/{id}",
223    request_body = CreateClassRequest,
224    responses((status = 200, description = "Class updated")),
225    tag = TAG
226)]
227pub async fn update(
228    Extension(user): Extension<AuthUser>,
229    State(ctx): State<AppContext>,
230    Path(id): Path<String>,
231    Json(payload): Json<CreateClassRequest>)
232     -> Result<Response> {
233    let _ = ensure_role(&user, &["teacher", "admin"])?;
234    let db = &ctx.db;
235    let now = chrono::Utc::now().timestamp();
236
237    let mut active: class::ActiveModel = class::Entity::find_by_id(&id)
238        .one(db)
239        .await
240        .map_err(|_| loco_rs::Error::InternalServerError)?
241        .ok_or(loco_rs::Error::NotFound)?
242        .into();
243
244    active.name = sea_orm::ActiveValue::Set(payload.name);
245    active.subject_area = sea_orm::ActiveValue::Set(payload.subject_area);
246    active.teacher_id = sea_orm::ActiveValue::Set(payload.teacher_id);
247    active.period = sea_orm::ActiveValue::Set(payload.period);
248    active.academic_year = sea_orm::ActiveValue::Set(payload.academic_year);
249    active.updated_at = sea_orm::ActiveValue::Set(now);
250
251    class::Entity::update(active).exec(db).await.map_err(|e| {
252        tracing::error!("Failed to update class: {:?}", e);
253        loco_rs::Error::InternalServerError
254    })?;
255
256    format::json(serde_json::json!({"status": "ok"}))
257}

318/// Enroll students in a class.
319#[utoipa::path(
320    post,
321    path = "/management/classes/{id}/students",
322    request_body = EnrollStudentsRequest,
323    responses((status = 200, description = "Students enrolled")),
324    tag = TAG
325)]
326pub async fn enroll_students(
327    Extension(user): Extension<AuthUser>,
328    State(ctx): State<AppContext>,
329    Path(id): Path<String>,
330    Json(payload): Json<EnrollStudentsRequest>)
331     -> Result<Response> {
332    let _ = ensure_role(&user, &["teacher", "admin"])?;
333    let db = &ctx.db;
334    let now = chrono::Utc::now().timestamp();
335    let mut count = 0;
336
337    for student_id in &payload.student_ids {
338        let active_model = class_enrollment::ActiveModel {
339            id: sea_orm::ActiveValue::NotSet,
340            class_id: sea_orm::ActiveValue::Set(id.clone()),
341            student_id: sea_orm::ActiveValue::Set(student_id.clone()),
342            enrolled_at: sea_orm::ActiveValue::Set(now),
343            active: sea_orm::ActiveValue::Set(true),
344            display_name: sea_orm::ActiveValue::Set(None),
345            metadata_json: sea_orm::ActiveValue::Set(None),
346        };
347
348        match class_enrollment::Entity::insert(active_model).exec(db).await {
349            Ok(_) => count += 1,
350            Err(e) => tracing::warn!("Failed to enroll student {}: {:?}", student_id, e),
351        }
352    }
353
354    format::json(serde_json::json!({"enrolled": count, "total": payload.student_ids.len()}))
355}

357/// Transfer a student out of a class (soft-delete).
358#[utoipa::path(
359    delete,
360    path = "/management/classes/{id}/students/{student_id}",
361    responses((status = 200, description = "Student transferred out")),
362    tag = TAG
363)]
364pub async fn transfer_student(
365    Extension(user): Extension<AuthUser>,
366    State(ctx): State<AppContext>,
367    Path((id, student_id)): Path<(String, String)>)
368     -> Result<Response> {
369    let _ = ensure_role(&user, &["teacher", "admin"])?;
370    let db = &ctx.db;
371
372    let enrollment = class_enrollment::Entity::find()
373        .filter(class_enrollment::Column::ClassId.eq(&id))
374        .filter(class_enrollment::Column::StudentId.eq(&student_id))
375        .filter(class_enrollment::Column::Active.eq(true))
376        .one(db)
377        .await
378        .map_err(|_| loco_rs::Error::InternalServerError)?
379        .ok_or(loco_rs::Error::NotFound)?;
380
381    let enrollment_id = enrollment.id;
382    let mut active: class_enrollment::ActiveModel = enrollment.into();
383    active.active = sea_orm::ActiveValue::Set(false);
384    if let Err(e) = active.update(db).await {
385        if e.to_string().contains("UNIQUE constraint failed") {
386            // Unique constraint failed because a matching historical inactive record already exists!
387            // We can safely hard-delete this duplicate active record instead.
388            class_enrollment::Entity::delete_by_id(enrollment_id).exec(db).await.map_err(|delete_err| {
389                tracing::error!("Failed to fallback delete student enrollment: {:?}", delete_err);
390                loco_rs::Error::InternalServerError
391            })?;
392        } else {
393            tracing::error!("Failed to transfer student: {:?}", e);
394            return Err(loco_rs::Error::InternalServerError);
395        }
396    }
397
398    format::json(serde_json::json!({"status": "transferred"}))
399}

478/// Self-enroll in a class using its 6-character code.
479///
480/// Requires a valid JWT in the Authorization header. The authenticated student
481/// is enrolled in the class matching the given code.
482#[derive(Debug, Deserialize, ToSchema)]
483pub struct EnrollByCodeRequest {
484    /// Optional display name override.
485    pub display_name: Option<String>,
486}
487
488#[utoipa::path(
489    post,
490    path = "/management/classes/code/{code}/enroll",
491    request_body = EnrollByCodeRequest,
492    responses(
493        (status = 200, description = "Enrolled"),
494        (status = 404, description = "Code not found"),
495        (status = 409, description = "Already enrolled")
496    ),
497    tag = TAG
498)]
499pub async fn enroll_by_code(
500    Extension(user): Extension<AuthUser>,
501    State(ctx): State<AppContext>,
502    Path(code): Path<String>,
503    Json(payload): Json<EnrollByCodeRequest>
504) -> Result<Response> {
505    // Student self-enrollment: any authenticated student may join a class by code.
506    // (No teacher/admin gate — this endpoint enrolls the calling user's own
507    // student profile, identified by the JWT's subject.)
508    let db = &ctx.db;
509    let now = chrono::Utc::now().timestamp();
510
511    // Find class by code
512    let class = class::Entity::find()
513        .filter(class::Column::ClassCode.eq(&code))
514        .one(db)
515        .await
516        .map_err(|_| loco_rs::Error::InternalServerError)?
517        .ok_or(loco_rs::Error::NotFound)?;
518
519    let student_id = user.id.to_string();
520
521    // Check if already enrolled
522    let existing = class_enrollment::Entity::find()
523        .filter(class_enrollment::Column::ClassId.eq(&class.id))
524        .filter(class_enrollment::Column::StudentId.eq(&student_id))
525        .filter(class_enrollment::Column::Active.eq(true))
526        .one(db)
527        .await
528        .map_err(|_| loco_rs::Error::InternalServerError)?;
529
530    if existing.is_some() {
531        return Err(Error::BadRequest("already enrolled in this class".into()));
532    }
533
534    // Enroll
535    let active_model = class_enrollment::ActiveModel {
536        id: sea_orm::ActiveValue::NotSet,
537        class_id: sea_orm::ActiveValue::Set(class.id.clone()),
538        student_id: sea_orm::ActiveValue::Set(student_id),
539        enrolled_at: sea_orm::ActiveValue::Set(now),
540        active: sea_orm::ActiveValue::Set(true),
541        display_name: sea_orm::ActiveValue::Set(payload.display_name),
542        metadata_json: sea_orm::ActiveValue::Set(None),
543    };
544
545    class_enrollment::Entity::insert(active_model)
546        .exec(db)
547        .await
548        .map_err(|e| {
549            tracing::error!("Failed to enroll student: {:?}", e);
550            loco_rs::Error::InternalServerError
551        })?;
552
553    format::json(serde_json::json!({
554        "status": "enrolled",
555        "class_id": class.id,
556        "class_name": class.name,
557        "student_id": user.id,
558    }))
