1//! Health check controller — `/api/v1/health` endpoint.
2//!
3//! Provides a DB-aware health check for load balancers and monitoring.
4
5use axum::extract::State;
6use loco_rs::prelude::*;
7
8pub fn routes() -> Routes {
9    Routes::new()
10        .prefix("/health")
11        .add("/", get(check))
12}
13
14/// Full health check with database connectivity.
15///
16/// Returns 200 `{"status":"ok",...}` if the database is reachable,
17/// or 503 `{"status":"degraded",...}` if it's not.
18pub async fn check(State(ctx): State<AppContext>) -> Result<Response> {
19    let db_ok = ctx.db.ping().await.is_ok();
20    let status = if db_ok { "ok" } else { "degraded" };
21    let http_code = if db_ok { 200 } else { 503 };
22
23    let body = serde_json::json!({
24        "status": status,
25        "service": "ankitov",
26        "version": env!("CARGO_PKG_VERSION"),
27        "db": db_ok,
28    });
29
30    Ok((
31        axum::http::StatusCode::from_u16(http_code).unwrap(),
32        [(axum::http::header::CONTENT_TYPE, "application/json")],
33        serde_json::to_string_pretty(&body).unwrap(),
34    )
35        .into_response())
36}
