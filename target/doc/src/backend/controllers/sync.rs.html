1//! Card-review sync WebSocket endpoint.
2//!
3//! Receives card review transactions from clients over a WebSocket and
4//! rebroadcasts them on a shared `tokio::sync::broadcast` channel for
5//! fan-out to other subscribers (e.g. live dashboards / the display wall).
6//! Each accepted transaction is acknowledged to the sender with `SYNC_ACK`.
7
8use axum::{extract::ws::{Message, WebSocket, WebSocketUpgrade}, response::IntoResponse, routing::get, Extension};
9use loco_rs::prelude::*;
10use serde::{Deserialize, Serialize};
11use tokio::sync::broadcast;
12
13/// One card review event as sent by a client over the sync WebSocket.
14#[derive(Clone, Serialize, Deserialize, Debug)]
15pub struct CardReviewTransaction {
16    /// Deck identifier the reviewed card belongs to.
17    pub deck_id: String,
18    /// Anki card identifier that was reviewed.
19    pub card_id: String,
20    /// Review grade (recall rating) assigned by the student.
21    pub grade: u8,
22    /// Unix timestamp in seconds (UTC) of the review event.
23    pub timestamp: i64,
24}
25
26/// WebSocket upgrade for `/sync/stream`.
27///
28/// Accepts the upgrade and hands the socket to `handle_websocket_loop`,
29/// forwarding the shared broadcast sender for rebroadcast.
30pub async fn sync_stream_handler(
31    ws: WebSocketUpgrade,
32    Extension(tx): Extension<broadcast::Sender<String>>,
33) -> impl IntoResponse {
34    ws.on_upgrade(|socket| handle_websocket_loop(socket, tx))
35}
36
37/// Receive loop for one connected sync client.
38///
39/// Reads text frames, parses each as a [`CardReviewTransaction`], and:
40/// 1. rebroadcasts the JSON payload on `tx` (subscribers get a copy), and
41/// 2. replies with a bare `SYNC_ACK` text frame.
42///
43/// Frames that are not valid JSON are silently dropped. The loop ends when
44/// the client disconnects or the ack write fails.
45async fn handle_websocket_loop(mut socket: WebSocket, tx: broadcast::Sender<String>) {
46    while let Some(Ok(frame)) = socket.recv().await {
47        if let Message::Text(text_payload) = frame {
48            if let Ok(transaction) = serde_json::from_str::<CardReviewTransaction>(&text_payload) {
49                let _ = tx.send(serde_json::to_string(&transaction).unwrap());
50                if socket.send(Message::Text("SYNC_ACK".into())).await.is_err() { break; }
51            }
52        }
53    }
54}
55
56/// Routes for the card-review sync WebSocket.
57pub fn routes() -> Routes {
58    Routes::new().add("/sync/stream", get(sync_stream_handler))
59}
