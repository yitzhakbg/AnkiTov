1//! Display leaderboard controller — **public, token-gated, no JWT**
2//! (spec §8.1).
3//!
4//! `GET /api/v1/display/class/<token>?window=week&metric=points&format=board|wall`
5//!
6//! The `<token>` is the unguessable per-class display token minted by the
7//! staff controller (`POST /management/leaderboard/<class_id>/display`) and
8//! stored in `class_display_boards.token`. Rotating the token invalidates
9//! the old link (same 404 code as a never-minted one — spec §8.6).
10//!
11//! The display is *read-only* and never returns `deeper` (spec §8.5).
12//! It auto-polls on the client every `leaderboard.display_refresh_secs`
13//! (spec §9.1); no server-side state is required to serve it.
14
15use crate::services::leaderboard::{self as lb, Metric, Window};
16use crate::services::leaderboard_aggregate as agg;
17use axum::extract::{Path, Query, State};
18use axum::routing::get;
19use loco_rs::app::AppContext;
20use loco_rs::prelude::*;
21use serde::Deserialize;
22use sea_orm::{ColumnTrait, EntityTrait, QueryFilter};

24const TAG: &str = "Leaderboard";
25
26pub fn routes() -> Routes {
27    Routes::new()
28        .prefix("/display")
29        .add("/class/:token", get(display_class_board))
30}

36/// Query-string parameters (spec §8.4).
37#[derive(Debug, Deserialize, utoipa::ToSchema)]
38pub struct DisplayQuery {
39    /// Window (spec §8.4).
40    pub window: Option<String>,
41    /// Metric (spec §8.4).
42    pub metric: Option<String>,
43    /// `board` (default) or `wall` (spec §8.1).
44    pub format: Option<String>,
45}
46
47fn default_window() -> Window { Window::Week }
48fn default_metric() -> Metric { Metric::Points }

54/// Resolve `<token>` → class + board; respond 404 for unknown/rotated.
55///
56/// `format=wall` returns the same board but with `entries` trimmed to
57/// the top-N (`per_page` default 10) and no `unranked` list — the
58/// projector wants a big-number view, not a full ranking (spec §8.1).
59///
60/// **No JWT**: the token is the credential. This route is *not* covered
61/// by `require_auth_for_management` (that middleware is path-scoped to
62/// `/api/v1/management/*` only).
63#[utoipa::path(
64    get,
65    path = "/display/class/{token}",
66    params(
67        ("token" = String, Path, description = "Per-class display token (unguessable)"),
68        ("window" = Option<String>, Query, description = "week | month | all"),
69        ("metric" = Option<String>, Query, description = "points (default) | compliance | volume"),
70        ("format" = Option<String>, Query, description = "board (default) | wall")
71    ),
72    responses(
73        (status = 200, description = "Class board", body = crate::services::leaderboard_aggregate::LeaderboardResponse),
74        (status = 404, description = "Unknown or rotated display token — same code, don't confirm validity (spec §8.6)")
75    ),
76    tag = TAG
77)]
78pub async fn display_class_board(
79    State(ctx): State<AppContext>,
80    Path(token): Path<String>,
81    Query(q): Query<DisplayQuery>,
82) -> Result<Response> {
83    // 1. Parse the query params with defaults.
84    let window = q
85        .window
86        .as_deref()
87        .and_then(|w| Window::parse(w).ok())
88        .unwrap_or(default_window());
89    let metric = q
90        .metric
91        .as_deref()
92        .and_then(|m| Metric::parse(m).ok())
93        .unwrap_or(default_metric());
94
95    // 2. Resolve the token → class_id.
96    let board = crate::models::entities::class_display_board::Entity::find()
97        .filter(crate::models::entities::class_display_board::Column::Token.eq(&token))
98        .filter(crate::models::entities::class_display_board::Column::Enabled.eq(true))
99        .one(&ctx.db)
100        .await
101        .map_err(|e| {
102            tracing::error!("display_class_board: token lookup failed: {e:?}");
103            loco_rs::Error::InternalServerError
104        })?
105        .ok_or_else(|| {
106            // Same 404 code for unknown AND rotated (spec §8.6).
107            loco_rs::Error::NotFound
108        })?;
109
110    let class_id = board.class_id.clone();
111    let class_name = agg::class_name(&ctx, &class_id).await;
112
113    // 3. Aggregate + rank.
114    let cohort = agg::aggregate_class_cohort(&ctx, &class_id, window).await?;
115    let min_cohort = 3usize; // spec §7.3 / leaderboard.min_cohort
116    let b = lb::rank(
117        metric,
118        window,
119        &cohort,
120        None, // no viewer on the display surface
121        min_cohort,
122        lb::PointsWeights::default(),
123    );
124
125    // 4. Build the envelope.
126    let scope = agg::Scope::Class;
127    let now = chrono::Utc::now().timestamp();
128    let response = agg::LeaderboardResponse {
129        scope,
130        class_id,
131        class_name,
132        metric: metric.as_str().to_string(),
133        window: window.as_str().to_string(),
134        generated_at: now,
135        total_ranked: b.total_ranked,
136        entries: b.entries,
137        unranked: b.unranked,
138        me: None, // display never has a viewer
139        deeper: None, // display never returns deeper (spec §8.5)
140    };
141
142    format::json(response)
143}
