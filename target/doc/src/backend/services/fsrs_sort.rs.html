1//! FSRS Card Sort — retrieve due cards from Anki SQLite and rank by urgency.
2//!
3//! Opens the `.anki2` database read-only (immutable mode) to query overdue
4//! review and learning cards filtered by note tags. Computes an approximate
5//! probability-of-recall (retrievability) for each card using the interval
6//! and overdue duration, then sorts **most urgent first** (lowest R).
7//!
8//! Also queries `revlog.time` to compute per-track average seconds-per-card
9//! for the capsule slicer's time-bound calculation.
10//!
11//! ## Approximate Recall Model
12//!
13//! For relative ordering (not exact FSRS prediction):
14//!
15//! ```text
16//! overdue_days = today - day_due
17//! if overdue_days <= 0:  R ≈ 0.95  (not yet due; already stable)
18//! else:                  R ≈ 0.90 × 0.5^(overdue_days / max(ivl, 1))
19//! ```
20//!
21//! This half-life decay model yields correct **relative** ordering:
22//! cards further past due with shorter intervals are most urgent.
23//! Exact FSRS state parsing from `cards.data` JSON is a v2 enhancement.
24

50/// A due card with its track tag and computed urgency.
51#[derive(Debug, Clone)]
52pub struct RankedCard {
53    pub card_id: i64,
54    pub note_id: i64,
55    /// The track tag this card matches (first matching tag).
56    pub track_tag: String,
57    /// Approximate probability of recall (0.0–1.0).
58    /// Lower = more urgent.
59    pub approximate_recall: f64,
60    /// Days overdue (0 = due today, positive = past due).
61    pub overdue_days: i32,
62}

64/// Result of the full fetch-and-sort operation.
65#[derive(Debug, Clone)]
66pub struct SortResult {
67    /// Cards sorted by ascending approximate_recall (most urgent first).
68    pub ranked: Vec<RankedCard>,
69    /// Total due cards per track tag.
70    pub track_counts: HashMap<String, usize>,
71    /// Estimated average seconds per card (cold-start or from revlog).
72    pub avg_seconds_per_card: f64,
73}

82/// Open the `.anki2` file and retrieve all due cards matching the given
83/// tags, sorted by FSRS urgency (lowest approximate recall first).
84///
85/// # Arguments
86/// * `anki2_path` — Path to the `.anki2` collection file.
87/// * `tags` — Track tags to filter cards by (matched via notes.tags LIKE).
88/// * `lookback_days` — Number of days to look back in revlog for per-card
89///   timing estimates (30-day window per spec).
90pub fn fetch_and_sort(
91    anki2_path: impl AsRef<Path>,
92    tags: &[String],
93    lookback_days: u32,
94) -> Result<SortResult, FsrsSortError> {
95    if tags.is_empty() {
96        return Ok(SortResult {
97            ranked: Vec::new(),
98            track_counts: HashMap::new(),
99            avg_seconds_per_card: crate::services::capsule_slicer::COLD_START_SECONDS_PER_CARD,
100        });
101    }
102
103    let path_str = anki2_path.as_ref().display().to_string();
104    let uri = format!("file:{}?immutable=1", path_str);
105    let conn = Connection::open_with_flags(
106        &uri,
107        OpenFlags::SQLITE_OPEN_READ_ONLY
108            | OpenFlags::SQLITE_OPEN_NO_MUTEX
109            | OpenFlags::SQLITE_OPEN_URI,
110    )
111    .map_err(|e| FsrsSortError::CannotOpen(path_str, e))?;
112
113    // ── Get collection creation timestamp ──────────────────────────────
114    let crt: i64 = conn
115        .query_row("SELECT crt FROM col LIMIT 1", [], |row| row.get(0))
116        .map_err(|e| FsrsSortError::QueryFailed("col.crt".into(), e))?;
117
118    let now_secs = std::time::SystemTime::now()
119        .duration_since(std::time::UNIX_EPOCH)
120        .unwrap()
121        .as_secs() as i64;
122    let today_day = (now_secs - crt) / 86400;
123
124    // ── Build tag filter ──────────────────────────────────────────────
125    // Anki stores tags space-separated in notes.tags.
126    // We escape single quotes and use LIKE for substring matching.
127    let escaped_tags: Vec<String> = tags
128        .iter()
129        .map(|t| format!("n.tags LIKE '%{}%'", t.replace('\'', "''")))
130        .collect();
131    let tag_filter = escaped_tags.join(" OR ");
132
133    // ── Query due cards ───────────────────────────────────────────────
134    // Anki queue semantics: -1 = suspended, -2 = user-buried, -3 = sched-buried,
135    // 0 = new, 1 = learning, 2 = review, 3 = day-learn.
136    // Only admit active cards (queue >= 0).
137    // type=2 (review): due is a day number → due if due <= today_day
138    // type=1 (learning): due is a Unix timestamp → due if due <= now_secs
139    // type=0 (new): always included
140    let sql = format!(
141        "SELECT c.id, c.nid, n.tags, c.ivl, c.type, c.due \
142         FROM cards c \
143         JOIN notes n ON c.nid = n.id \
144         WHERE c.queue >= 0 \
145           AND ({tag_filter}) \
146           AND ( \
147             (c.type = 2 AND c.due <= {today_day}) \
148             OR (c.type = 1 AND c.due <= {now_secs}) \
149             OR (c.type = 0) \
150           ) \
151         ORDER BY c.id"
152    );
153
154    let mut stmt = conn
155        .prepare(&sql)
156        .map_err(|e| FsrsSortError::QueryFailed("cards query prep".into(), e))?;
157
158    let rows = stmt
159        .query_map([], |row| {
160            Ok((
161                row.get::<_, i64>(0)?,   // c.id
162                row.get::<_, i64>(1)?,   // c.nid
163                row.get::<_, String>(2)?, // n.tags
164                row.get::<_, i64>(3)?,   // c.ivl
165                row.get::<_, i64>(4)?,   // c.type
166                row.get::<_, i64>(5)?,   // c.due
167            ))
168        })
169        .map_err(|e| FsrsSortError::QueryFailed("cards query exec".into(), e))?;
170
171    // ── Compute approximate recall and assign track tags ──────────────
172    let mut track_counts: HashMap<String, usize> = HashMap::new();
173    let mut cards: Vec<RankedCard> = Vec::new();
174
175    for row in rows {
176        let (card_id, note_id, note_tags, ivl, card_type, due) = row
177            .map_err(|e| FsrsSortError::QueryFailed("row read".into(), e))?;
178
179        // Find which of our track tags matches this note
180        let matched_tag = match find_matching_tag(&note_tags, tags) {
181            Some(t) => t,
182            None => continue, // shouldn't happen with our filter, but be safe
183        };
184
185        let overdue_days = compute_overdue_days(card_type, due, today_day, now_secs, crt);
186        let approximate_recall = compute_approximate_recall(overdue_days, ivl);
187
188        *track_counts.entry(matched_tag.clone()).or_insert(0) += 1;
189
190        cards.push(RankedCard {
191            card_id,
192            note_id,
193            track_tag: matched_tag,
194            approximate_recall,
195            overdue_days,
196        });
197    }
198
199    // ── Sort by urgency (lowest recall first), then overdue_days desc,
200    //     then card_id ascending (deterministic tie-break) ──
201    cards.sort_by(|a, b| {
202        a.approximate_recall
203            .partial_cmp(&b.approximate_recall)
204            .unwrap_or(std::cmp::Ordering::Equal)
205            .then_with(|| b.overdue_days.cmp(&a.overdue_days))
206            .then_with(|| a.card_id.cmp(&b.card_id))
207    });
208
209    // ── Compute per-track average time from revlog ────────────────────
210    let avg_seconds = compute_avg_time_per_card(&conn, tags, lookback_days)
211        .unwrap_or(crate::services::capsule_slicer::COLD_START_SECONDS_PER_CARD);
212
213    Ok(SortResult {
214        ranked: cards,
215        track_counts,
216        avg_seconds_per_card: avg_seconds,
217    })
218}

224/// Find the first track tag that appears in the note's space-separated tags.
225fn find_matching_tag(note_tags: &str, track_tags: &[String]) -> Option<String> {
226    // Anki tags are space-separated. Split and check for membership.
227    let note_tag_set: std::collections::HashSet<&str> =
228        note_tags.split_whitespace().collect();
229
230    for tag in track_tags {
231        if note_tag_set.contains(tag.as_str()) {
232            return Some(tag.clone());
233        }
234    }
235    None
236}

238/// Compute days overdue relative to the due date.
239fn compute_overdue_days(
240    card_type: i64,
241    due: i64,
242    today_day: i64,
243    now_secs: i64,
244    _crt: i64,
245) -> i32 {
246    match card_type {
247        2 => {
248            // Review card: `due` is a day number
249            (today_day - due) as i32
250        }
251        1 => {
252            // Learning card: `due` is a Unix timestamp in seconds
253            let seconds_overdue = now_secs - due;
254            (seconds_overdue / 86400) as i32
255        }
256        _ => {
257            // New cards (type=0): treat as "due now" → overdue_days = 0
258            0
259        }
260    }
261}

263/// Compute approximate probability of recall using a half-life decay model.
264///
265/// ```text
266/// R ≈ 0.90 × 0.5^(overdue_days / ivl)
267/// ```
268///
269/// This is a simplification of the FSRS retrievability formula:
270/// `R = (1 + 19 × t / S)^(-1)` where S ≈ ivl and t = elapsed_days.
271///
272/// For relative ordering, half-life decay is sufficient. Exact FSRS state
273/// parsing (from `cards.data` JSON) is a v2 enhancement.
274/// Compute approximate probability of recall using a half-life decay model.
275///
276/// ```text
277/// R ≈ 0.90 × 0.5^(overdue_days / ivl)
278/// ```
279pub fn compute_approximate_recall(overdue_days: i32, ivl: i64) -> f64 {
280    if overdue_days <= 0 {
281        // Card is not yet due — high recall
282        return 0.95;
283    }
284
285    let effective_ivl = ivl.max(1) as f64;
286    let decay_exponent = overdue_days as f64 / effective_ivl;
287
288    // Base recall at due time: 0.90 (FSRS default desired retention)
289    0.90 * (0.5_f64).powf(decay_exponent)
290}

292/// Compute average seconds per card from revlog history, grouped by track tags.
293///
294/// Returns the overall average across all matching tracks, or `None` if no
295/// historical data exists (caller should fall back to cold-start default).
296fn compute_avg_time_per_card(
297    conn: &Connection,
298    tags: &[String],
299    lookback_days: u32,
300) -> Option<f64> {
301    let cutoff_ms = std::time::SystemTime::now()
302        .duration_since(std::time::UNIX_EPOCH)
303        .unwrap()
304        .as_millis() as i64
305        - (lookback_days as i64 * 86_400_000); // milliseconds
306
307    // Build tag filter for revlog query
308    let escaped_tags: Vec<String> = tags
309        .iter()
310        .map(|t| format!("n.tags LIKE '%{}%'", t.replace('\'', "''")))
311        .collect();
312    let tag_filter = escaped_tags.join(" OR ");
313
314    let sql = format!(
315        "SELECT AVG(r.time) / 1000.0 \
316         FROM revlog r \
317         JOIN cards c ON r.cid = c.id \
318         JOIN notes n ON c.nid = n.id \
319         WHERE r.id >= {cutoff_ms} \
320           AND ({tag_filter}) \
321           AND r.time > 0"
322    );
323
324    let result: Option<f64> = conn
325        .query_row(&sql, [], |row| row.get(0))
326        .ok()
327        .flatten();
328
329    // If we have data, return it; otherwise None triggers cold-start
330    result.filter(|&t| t > 0.0 && t.is_finite())
331}

366        // More overdue + shorter interval = more urgent (lower R)
367        let r1 = compute_approximate_recall(5, 5);  // overdue 5, ivl 5
368        let r2 = compute_approximate_recall(5, 50); // overdue 5, ivl 50
369        let r3 = compute_approximate_recall(20, 50); // overdue 20, ivl 50
370
371        // r1 (0.45) < r2 (0.84) < r3 (0.68)
372        assert!(r1 < r2); // shorter interval → lower recall
373        assert!(r3 < r2); // more overdue → lower recall
374    }
