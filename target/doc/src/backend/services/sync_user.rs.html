1//! Per-student Anki sync-server user provisioning.
2//!
3//! # Why this exists
4//!
5//! The self-hosted Anki sync server (`ankitov-sync-server`, Anki `--syncserver`)
6//! is a *multi-user* sync hub: each student connects with their own
7//! `username:password` and the server keeps their collection isolated. The
8//! server discovers its users from `SYNC_USER1..N` env vars, which the
9//! container entrypoint populates from a shared file
10//! (`/etc/ankitov/sync-users.env`, one `username:password` per line) that both
11//! the **backend** and **sync-server** containers mount.
12//!
13//! This module is the *writer* of that file. When a student is imported/enrolled
14//! it:
15//!
16//! 1. Derives a **unique Anki profile name from the student's name** —
17//!    `first.last`, lowercased, non-alphanumerics stripped, with a `-<id>`
18//!    suffix if another student already resolved to the same base name.
19//! 2. Sets the sync **password = the student's numeric id** (the id from the
20//!    imported student list, i.e. the `users.id` primary key).
21//! 3. Writes `profile_name:<id>` into the shared `sync-users.env`.
22//!
23//! The same `profile_name` / `id` pair is the credential the Selkies stream
24//! container uses to log in as that student (stream `PASSWD` = the student id),
25//! so a single identity map drives both "sync your deck on your device" and
26//! "pick up where you left off on the streamed device."
27//!
28//! # Safety rules
29//!
30//! | Rule | Enforcement |
31//! |------|-------------|
32//! | Profile names are unique in the sync file | base name + `-<id>` suffix on collision; the numeric `id` (the password) is always unique regardless |
33//! | Sync password is never the real login password | password = numeric id only; the *sync* credential is separate from the AnkiTov login password |
34//! | File is a 600-mode file, written atomically | temp file + rename; `set_permissions(0o600)` |
35//! | Never a write to the Anki DB | this only touches the credential file, never the `.anki2` collection |

45#[derive(Error, Debug)]
46pub enum SyncUserError {
47    #[error("no full_name and no username for user {0}; cannot derive a profile name")]
48    NoName(i64),
49    #[error("could not write sync user file: {0}")]
50    Io(std::io::Error),
51    #[error("{0}")]
52    Other(String),
53}

55/// The shared file path. The backend container mounts the sync-server's
56/// `/etc/ankitov` volume at the same path; overridable via `ANKITOV_SYNC_USER_FILE`.
57pub const DEFAULT_SYNC_USER_FILE: &str = "/etc/ankitov/sync-users.env";
58
59/// Resolve the file path from env (or default).
60pub fn sync_user_file() -> std::path::PathBuf {
61    std::env::var("ANKITOV_SYNC_USER_FILE")
62        .map(std::path::PathBuf::from)
63        .unwrap_or_else(|_| std::path::PathBuf::from(DEFAULT_SYNC_USER_FILE))
64}

66/// Derive a base Anki profile name from a full name: `first.last`,
67/// lowercased, non-alphanumeric characters (except `.`) removed.
68///
69/// Mirrors `derive_username` in `management/teachers.rs` so a student's
70/// AnkiTov username and Anki profile stay in the same family.
71///
72/// # Examples
73/// ```
74/// assert_eq!(super::base_profile_name("Alice Smith"), "alice.smith");
75/// assert_eq!(super::base_profile_name("  Marcus  Nguyen  "), "marcus.nguyen");
76/// assert_eq!(super::base_profile_name("O'Neil"), "oneil");
77/// assert_eq!(super::base_profile_name("Solo"), "solo");
78/// ```
79pub fn base_profile_name(full_name: &str) -> String {
80    let parts: Vec<&str> = full_name
81        .trim()
82        .split_whitespace()
83        .filter(|p| !p.is_empty())
84        .collect();
85    let first = parts.first().copied().unwrap_or("");
86    let last = if parts.len() >= 2 {
87        parts.last().copied().unwrap_or(first)
88    } else {
89        ""
90    };
91
92    let mut out = String::new();
93    for c in first.chars() {
94        out.push(c.to_ascii_lowercase());
95    }
96    if !last.is_empty() {
97        if !out.is_empty() {
98            out.push('.');
99        }
100        for c in last.chars() {
101            out.push(c.to_ascii_lowercase());
102        }
103    }
104    let cleaned: String = out
105        .chars()
106        .filter(|c| c.is_ascii_alphanumeric() || *c == '.')
107        .collect();
108    let trimmed = cleaned.trim_matches('.').to_string();
109    if trimmed.is_empty() {
110        "user".into()
111    } else {
112        trimmed
113    }
114}

116/// Produce a *unique* profile name for the given user, deduping against
117/// existing students.
118///
119/// Starts from the name-derived base. If another student already has a row
120/// whose username equals the base (i.e. the bare name is taken), append the
121/// student's numeric `id` as a `-<id>` suffix so the two never collide. The
122/// password (also `id`) is the true uniqueness key, so even a base-name
123/// collision still yields a distinct sync identity; the suffix keeps the
124/// *name* readable and unambiguous in the UI.
125pub async fn unique_profile_name(
126    db: &DatabaseConnection,
127    user_id: i64,
128    full_name: &str,
129) -> Result<String, SyncUserError> {
130    let base = base_profile_name(full_name);
131
132    // One filter on the base username, then exclude ourselves in Rust (an id
133    // exclusion would need a second predicate, and `Select::filter` takes a
134    // condition directly — chain a second `.filter(..)` if that ever changes).
135    let candidates = user::Entity::find()
136        .filter(user::Column::Username.eq(base.clone()))
137        .all(db)
138        .await
139        .map_err(|e| SyncUserError::Other(e.to_string()))?;
140
141    let others = candidates.iter().filter(|u| u.id != user_id).count();
142
143    if others > 0 {
144        Ok(format!("{base}-{user_id}"))
145    } else {
146        Ok(base)
147    }
148}

150/// Write a single student's sync credential into the shared file.
151///
152/// `user_id` (the student's numeric id) becomes the *password*. Adds-or-replaces
153/// the line for this student's profile name, preserving every other student's
154/// line. Returns the profile name that was written.
155pub async fn upsert_sync_user(
156    db: &DatabaseConnection,
157    user_id: i64,
158    full_name: &str,
159) -> Result<String, SyncUserError> {
160    let profile = unique_profile_name(db, user_id, full_name).await?;
161    let line = format!("{}:{}", profile, user_id);
162    let path = sync_user_file();
163    upsert_line(&path, &line)?;
164    Ok(profile)
165}

167/// Add-or-replace a `username:password` line, preserving all other lines.
168fn upsert_line(path: &std::path::Path, line: &str) -> Result<(), SyncUserError> {
169    let target_user: &str = line.split(':').next().unwrap_or("");
170    let existing = std::fs::read_to_string(path).unwrap_or_default();
171    let mut out = String::new();
172    let mut replaced = false;
173    for l in existing.lines() {
174        let l = l.trim();
175        if l.is_empty() || l.starts_with('#') {
176            out.push_str(l);
177            out.push('\n');
178            continue;
179        }
180        let existing_user = l.split(':').next().unwrap_or("");
181        if existing_user == target_user {
182            out.push_str(line);
183            out.push('\n');
184            replaced = true;
185        } else {
186            out.push_str(l);
187            out.push('\n');
188        }
189    }
190    if !replaced {
191        out.push_str(line);
192        out.push('\n');
193    }
194    write_atomic(path, &out)
195}

197/// Rewrite the whole file from a slice of `(profile_name, user_id)` pairs,
198/// de-duplicated by profile name and sorted for deterministic output. Used by
199/// the batch re-provisioning endpoint so a full import can be (re)written in
200/// one shot.
201pub async fn rewrite_sync_user_file(
202    path: &std::path::Path,
203    pairs: &[(String, i64)],
204) -> Result<(), SyncUserError> {
205    let mut seen: std::collections::HashSet<String> = Default::default();
206    let mut out = String::new();
207    for (u, id) in pairs {
208        if !seen.insert(u.clone()) {
209            continue;
210        }
211        out.push_str(&format!("{}:{}", u, id));
212        out.push('\n');
213    }
214    write_atomic(path, &out)
215}

217/// Write `content` to `path` atomically (temp file + rename) with 0600 mode.
218fn write_atomic(path: &std::path::Path, content: &str) -> Result<(), SyncUserError> {
219    if let Some(parent) = path.parent() {
220        std::fs::create_dir_all(parent)
221            .map_err(SyncUserError::Io)?;
222    }
223    let mut tmp = path.to_path_buf();
224    let name = path
225        .file_name()
226        .and_then(|n| n.to_str())
227        .unwrap_or("sync-users.env");
228    let suffix = std::process::id();
229    tmp.push(format!(".{}.{}.tmp", name, suffix));
230
231    std::fs::write(&tmp, content).map_err(SyncUserError::Io)?;
232    #[cfg(unix)]
233    {
234        use std::os::unix::fs::PermissionsExt;
235        let _ = std::fs::set_permissions(&tmp, std::fs::Permissions::from_mode(0o600));
236    }
237    std::fs::rename(&tmp, path).map_err(SyncUserError::Io)?;
238    Ok(())
239}
