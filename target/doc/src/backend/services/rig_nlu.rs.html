1//! Rig NLU Pipeline — Natural Language Understanding via rig-core.
2//!
3//! Accepts free-form dashboard chatbox queries and returns structured intents
4//! for downstream execution. Falls back to keyword matching when the LLM is
5//! unavailable, the budget gate rejects the request, or the response is invalid.
6//!
7//! # Provider selection
8//!
9//! Two providers are supported, selected via the `ANKITOV_NLU_PROVIDER` env var:
10//!
11//! - `freetoken` (default when set): points rig at a local FreeToken instance
12//!   (`http://localhost:1919/v1`) serving the OpenAI-compatible chat completions
13//!   API. Zero cost, zero internet dependency, school-local inference.
14//! - `deepseek` (legacy): the original cloud DeepSeek path (`deepseek-chat`).
15//! - unset: defaults to the local FreeToken endpoint if reachable, otherwise
16//!   falls back to DeepSeek cloud.
17//!
18//! The FreeToken endpoint, API key, and model are overridable via:
19//!   `ANKITOV_FREETOKEN_BASE_URL` (default `http://localhost:1919/v1`)
20//!   `ANKITOV_FREETOKEN_API_KEY` (default `freetoken` — FreeToken doesn't auth)
21//!   `ANKITOV_FREETOKEN_MODEL`  (default `Qwen3.6-35B-A3B`)
22//!
23//! The legacy DeepSeek model is overridable via:
24//!   `ANKITOV_RIG_MODEL` (default `deepseek-chat`)
25
26use rig::client::{CompletionClient, ProviderClient};
27use rig::completion::Prompt;
28use rig::providers::deepseek;
29use rig::providers::openai;
30use serde::{Deserialize, Serialize};

32/// Structured intent extracted from natural language.
33#[derive(Debug, Clone, Serialize, Deserialize)]
34pub struct NluIntent {
35    pub intent: String,
36    pub subject: Option<String>,
37    pub time_window_days: Option<i64>,
38    pub summary: String,
39}
40
41/// System preamble that instructs the LLM to output structured JSON.
42const NLU_PREAMBLE: &str = r#"You are an NLU intent parser for an educational analytics dashboard called AnkiTov.
43Your job is to analyze a teacher/administrator's question about student performance
44and extract structured data.
45
46## Available Intents
47- struggling_students — "who's having trouble?", "which students are failing?"
48- retention_drop — "practice dropped off", "retention declined recently"
49- deck_health — "how are my decks doing?", "overall health check"
50- recent_exceptions — "show me recent alerts", "any problems?"
51- missed_practice — "who missed practice", "who hasn't studied", "absent", "missed in the past 2 days", "consecutive missed days", "not practiced"
52- count_question — "how many students are there?", "how many classes/decks/cards?", "total number of X" — counting/enumeration questions
53- unknown — the question doesn't match any known intent
54
55## Subjects (extract from question if mentioned)
56math, algebra, geometry, biology, physics, chemistry, vocab, 7th grade, 8th grade, 9th grade
57
58## Time Windows (extract from question if mentioned)
59- "past week" / "last week" → 7
60- "past two weeks" / "last two weeks" → 14
61- "past month" / "last month" → 30
62- null if no timeframe mentioned
63
64## Output Format
65Respond with ONLY a single JSON object. No markdown, no explanation, no backticks.
66{"intent":"<intent>","subject":<string|null>,"time_window_days":<number|null>,"summary":"<concise human-readable summary>"}"#;

68/// Try LLM-based NLU first, then fall back to keyword matching.
69///
70/// Provider selection (env `ANKITOV_NLU_PROVIDER`):
71/// - `freetoken` → local FreeToken instance (no API key, no internet)
72/// - `deepseek`  → cloud DeepSeek API (legacy, requires `DEEPSEEK_API_KEY`)
73/// - unset       → try FreeToken first, fall back to DeepSeek, then keyword
74pub async fn parse_query(query: &str) -> NluIntent {
75    let provider = std::env::var("ANKITOV_NLU_PROVIDER").unwrap_or_default();
76
77    // ── FreeToken path ──────────────────────────────────────────────────
78    if provider == "freetoken" {
79        match tokio::time::timeout(
80            std::time::Duration::from_secs(3),
81            try_freetoken_nlu(query),
82        )
83        .await
84        {
85            Ok(Ok(intent)) => {
86                tracing::info!(
87                    intent = %intent.intent,
88                    subject = ?intent.subject,
89                    "FreeToken NLU parsed query successfully"
90                );
91                return intent;
92            }
93            Ok(Err(e)) => {
94                tracing::warn!("FreeToken NLU failed ({e}), falling back to keyword parser");
95                return keyword_fallback(query);
96            }
97            Err(_) => {
98                tracing::warn!("FreeToken NLU timed out (3s), falling back to keyword parser");
99                return keyword_fallback(query);
100            }
101        }
102    }
103
104    if provider == "deepseek" {
105        match tokio::time::timeout(
106            std::time::Duration::from_secs(5),
107            try_rig_nlu(query),
108        )
109        .await
110        {
111            Ok(Ok(intent)) => {
112                tracing::info!(
113                    intent = %intent.intent,
114                    subject = ?intent.subject,
115                    "Rig NLU parsed query successfully"
116                );
117                return intent;
118            }
119            Ok(Err(e)) => {
120                tracing::warn!("Rig NLU failed ({e}), falling back to keyword parser");
121                return keyword_fallback(query);
122            }
123            Err(_) => {
124                tracing::warn!("Rig NLU timed out (5s), falling back to keyword parser");
125                return keyword_fallback(query);
126            }
127        }
128    }
129
130    // Default or unset or "keyword" → instant, deterministic zero-latency keyword fallback
131    keyword_fallback(query)
132}

134/// Attempt to parse the query using a local FreeToken instance.
135///
136/// FreeToken serves an OpenAI-compatible chat completions API at a configurable
137/// base URL. No API key is required for local inference.
138///
139/// Environment variables:
140///   `ANKITOV_FREETOKEN_BASE_URL` — default `http://localhost:1919/v1`
141///   `ANKITOV_FREETOKEN_MODEL`    — default `Qwen3.6-35B-A3B`
142async fn try_freetoken_nlu(query: &str) -> Result<NluIntent, String> {
143    let base_url = std::env::var("ANKITOV_FREETOKEN_BASE_URL")
144        .unwrap_or_else(|_| "http://localhost:1919/v1".into());
145    let model = std::env::var("ANKITOV_FREETOKEN_MODEL")
146        .unwrap_or_else(|_| "Qwen3.6-35B-A3B".into());
147
148    let client = openai::CompletionsClient::builder()
149        .base_url(&base_url)
150        .api_key("freetoken") // FreeToken doesn't validate API keys
151        .build()
152        .map_err(|e| format!("Failed to create FreeToken client: {e}"))?;
153
154    let agent = client
155        .agent(&model)
156        .preamble(NLU_PREAMBLE)
157        .temperature(0.0) // deterministic for intent parsing
158        .build();
159
160    let user_prompt = format!("User question: \"{query}\"");
161    let response = agent
162        .prompt(&user_prompt)
163        .await
164        .map_err(|e| format!("FreeToken call failed: {e}"))?;
165
166    // Parse the JSON response — strip any accidental markdown fences
167    let json_str = response
168        .trim()
169        .trim_start_matches("```json")
170        .trim_start_matches("```")
171        .trim_end_matches("```")
172        .trim();
173
174    let intent: NluIntent = serde_json::from_str(json_str)
175        .map_err(|e| format!("JSON parse error: {e}. Raw response: {json_str}"))?;
176
177    // Basic validation
178    let valid_intents = [
179        "struggling_students",
180        "retention_drop",
181        "deck_health",
182        "recent_exceptions",
183        "missed_practice",
184        "count_question",
185        "unknown",
186    ];
187    if !valid_intents.contains(&intent.intent.as_str()) {
188        return Err(format!("Unknown intent: {}", intent.intent));
189    }
190
191    Ok(intent)
192}

194/// Attempt to parse the query using rig-core + DeepSeek.
195async fn try_rig_nlu(query: &str) -> Result<NluIntent, String> {
196    // Allow overriding the model via env
197    let model = std::env::var("ANKITOV_RIG_MODEL").unwrap_or_else(|_| "deepseek-chat".into());
198
199    let client = deepseek::Client::from_env()
200        .map_err(|e| format!("Failed to create DeepSeek client: {e}"))?;
201
202    let agent = client
203        .agent(&model)
204        .preamble(NLU_PREAMBLE)
205        .temperature(0.0) // deterministic for intent parsing
206        .build();
207
208    let user_prompt = format!("User question: \"{query}\"");
209    let response = agent.prompt(&user_prompt).await.map_err(|e| format!("LLM call failed: {e}"))?;
210
211    // Parse the JSON response — strip any accidental markdown fences
212    let json_str = response.trim().trim_start_matches("```json").trim_start_matches("```").trim_end_matches("```").trim();
213
214    let intent: NluIntent =
215        serde_json::from_str(json_str).map_err(|e| format!("JSON parse error: {e}. Raw response: {json_str}"))?;
216
217    // Basic validation
218    let valid_intents = ["struggling_students", "retention_drop", "deck_health", "recent_exceptions", "missed_practice", "count_question", "unknown"];
219    if !valid_intents.contains(&intent.intent.as_str()) {
220        return Err(format!("Unknown intent: {}", intent.intent));
221    }
222
223    Ok(intent)
224}

226/// Keyword-based fallback parser — identical logic to the original ask controller.
227fn keyword_fallback(q: &str) -> NluIntent {
228    let lower = q.to_lowercase();
229
230    // Intent detection
231    let trouble_words = ["trouble", "struggl", "failing", "behind", "low retention", "worst"];
232    let drop_words = ["dropped off", "fell off", "decline", "decreased", "getting worse", "worse over", "past two weeks", "past week", "lately"];
233    let health_words = ["deck health", "how are my decks", "decks doing", "overall health", "status of"];
234    let exception_words = ["exceptions", "problems", "alerts", "warnings", "flagged"];
235    let missed_words = ["miss", "practice", "haven't studied", "not studied", "studied", "absent", "consecutive", "haven't practiced", "not practiced", "lazy"];
236
237    // Count/enumeration questions: "how many X are there?", "total number of X",
238    // "number of X", "how many X do I have?". Must be checked BEFORE the
239    // struggle/miss/deck-health words because "how many students are having
240    // trouble?" is a count question, not a struggle question.
241    let is_count = (lower.contains("how many") || lower.contains("how much")
242        || lower.contains("total number") || lower.contains("total count")
243        || lower.contains("number of") || lower.contains("count of")
244        || lower.contains("how many") || lower.contains("give me the number"))
245        && (lower.contains("student") || lower.contains("teacher")
246            || lower.contains("class") || lower.contains("deck")
247            || lower.contains("card") || lower.contains("module")
248            || lower.contains("practice plan") || lower.contains("profile"));
249
250    // Deck / upload / distribution actions → "decks" (handled by the fallback,
251    // which routes the user to the Decks screen). Checked before the health
252    // words so "upload a deck" / "distribute the deck" don't get eaten by
253    // "status of" / "how are my decks".
254    let deck_action_words = ["upload", "distribut", "import", "add a deck", "new deck"];
255    let is_deck_action = deck_action_words.iter().any(|w| lower.contains(w));
256
257    let intent = if is_count {
258        "count_question"
259    } else if is_deck_action {
260        "decks"
261    } else if trouble_words.iter().any(|w| lower.contains(w)) {
262        "struggling_students"
263    } else if missed_words.iter().any(|w| lower.contains(w)) {
264        "missed_practice"
265    } else if drop_words.iter().any(|w| lower.contains(w)) {
266        "retention_drop"
267    } else if health_words.iter().any(|w| lower.contains(w)) {
268        "deck_health"
269    } else if exception_words.iter().any(|w| lower.contains(w)) {
270        "recent_exceptions"
271    } else {
272        "unknown"
273    };
274
275    // Subject extraction
276    let subjects = [
277        "math", "algebra", "geometry", "biology", "physics", "chemistry",
278        "vocab", "vocabulary", "spanish", "french", "history", "science",
279        "7th grade", "8th grade", "9th grade", "grade 7", "grade 8", "grade 9",
280    ];
281    let subject = subjects.iter().find(|s| lower.contains(*s)).map(|s| s.to_string());
282
283    // Time window
284    let mut time_window_days = if lower.contains("past two weeks") || lower.contains("last two weeks") || lower.contains("2 weeks") {
285        Some(14)
286    } else if lower.contains("past week") || lower.contains("last week") || lower.contains("1 week") {
287        Some(7)
288    } else if lower.contains("past month") || lower.contains("last month") {
289        Some(30)
290    } else if lower.contains("past 3 days") || lower.contains("last 3 days") {
291        Some(3)
292    } else {
293        None
294    };
295
296    if intent == "missed_practice" && time_window_days.is_none() {
297        if lower.contains("two consecutive") || lower.contains("2 consecutive") || lower.contains("two days") || lower.contains("2 days") {
298            time_window_days = Some(2);
299        } else if lower.contains("three consecutive") || lower.contains("3 consecutive") || lower.contains("three days") || lower.contains("3 days") {
300            time_window_days = Some(3);
301        } else if lower.contains("four consecutive") || lower.contains("4 consecutive") || lower.contains("four days") || lower.contains("4 days") {
302            time_window_days = Some(4);
303        } else {
304            time_window_days = Some(2); // default missed practice query to 2 days
305        }
306    }
307
308    let summary = format!("[keyword] Intent: {intent}, Subject: {:?}", subject);
309
310    NluIntent {
311        intent: intent.to_string(),
312        subject,
313        time_window_days,
314        summary,
315    }
316}
