1//! Zone 2: AnkiConnect Direct HTTP Client.
2//!
3//! ## Purpose
4//!
5//! Calls Anki's JSON-RPC API at `localhost:8765` directly from Rust via
6//! `reqwest`. Every write operation goes through Anki's own scheduler,
7//! guaranteeing SRS algorithm correctness. Sync protocol, .apkg handling,
8//! and live scheduling queries all flow through this client.
9//!
10//! ## No MCP Server in Production
11//!
12//! This replaces the `anki-mcp-server` NestJS/TypeScript MCP process that
13//! was suitable for development-time Goose agent interaction. At runtime,
14//! AnkiTov's Rust backend calls AnkiConnect HTTP directly — eliminating a
15//! ~80–120 MB Node.js process, a 3-hop transport chain, and a TypeScript
16//! supply chain from the production container.
17//!
18//! ## Safety Rules
19//!
20//! 1. **Concurrency = 1:** AnkiConnect is single-threaded. All calls are
21//!    serialized via `tokio::sync::Mutex` to prevent blocking or timeouts.
22//! 2. **Action whitelist:** Only pre-approved actions are permitted. Any
23//!    action not in the whitelist returns `Error::ActionNotAllowed` at call
24//!    time — not at configuration time. Defense-in-depth against accidents.
25//! 3. **Scheduler-safe:** All writes use Anki's state machine. Direct SQLite
26//!    writes are never used for scheduling operations.
27//!
28//! ## Endpoint Reference
29//!
30//! | Category | Actions |
31//! |----------|---------|
32//! | **Read-only** | `deckNames`, `getDeckStats`, `findCards`, `findNotes`, `cardsInfo`, `notesInfo`, `getDueCards`, `version` |
33//! | **Live** | `getDueCards`, `getDeckStats` |
34//! | **Sync** | `sync`, `syncLogin`, `syncRegister` |
35//! | **Write (safe)** | `suspend`, `unsuspend`, `setDueDate`, `exportPackage`, `importPackage` |
36//!
37//! ## Usage
38//!
39//! ```rust,ignore
40//! use crate::services::anki_connect::{AnkiConnectClient, AnkiConnectError};
41//!
42//! let client = AnkiConnectClient::new(8765, None);
43//! let stats = client.get_deck_stats("My Deck").await?;
44//! ```

115// ---------------------------------------------------------------------------
116// Error types
117// ---------------------------------------------------------------------------
118
119/// Errors specific to the AnkiConnect client layer.
120#[derive(Debug, thiserror::Error)]
121pub enum AnkiConnectError {
122    /// AnkiConnect returned a JSON-RPC error field.
123    #[error("AnkiConnect RPC error: {0}")]
124    RpcError(serde_json::Value),
125
126    /// The requested action is not in the approved whitelist.
127    #[error("action not in whitelist: {0}")]
128    ActionNotAllowed(String),
129
130    /// Anki is not running or AnkiConnect is unreachable.
131    #[error("cannot connect to AnkiConnect at localhost:{0} — is Anki running?")]
132    Unreachable(u16),
133
134    /// HTTP-level error (refused connection, timeout, etc.).
135    #[error("HTTP error: {0}")]
136    HttpError(#[from] reqwest::Error),
137
138    /// Serialization or deserialization failure.
139    #[error("JSON error: {0}")]
140    JsonError(#[from] serde_json::Error),
141
142    /// The API version returned by `version` does not match v6.
143    #[error("AnkiConnect version mismatch: got {0}, expected 6")]
144    VersionMismatch(i64),
145
146    /// No `result` field in the AnkiConnect response.
147    #[error("no result in AnkiConnect response — possible protocol error")]
148    NoResult,
149}
150
151impl AnkiConnectResponse {
152    /// Convert the JSON-RPC response into a typed result or return the error.
153    pub fn into_result<T: serde::de::DeserializeOwned>(self) -> Result<T, AnkiConnectError> {
154        if let Some(err) = self.error {
155            return Err(AnkiConnectError::RpcError(err));
156        }
157        let result = self
158            .result
159            .ok_or(AnkiConnectError::NoResult)?;
160        let value: T = serde_json::from_value(result)?;
161        Ok(value)
162    }
163}

169/// AnkiConnect Direct HTTP client.
170///
171/// Constructed with a port and optional API key. All calls are serialized
172/// through a `tokio::sync::Mutex` (concurrency = 1). Only whitelisted
173/// actions can be invoked.
174///
175/// # Example
176///
177/// ```rust,ignore
178/// let client = AnkiConnectClient::new(8765, Some("my-api-key"));
179///
180/// // List all deck names
181/// let decks = client.deck_names().await?;
182///
183/// // Get live due count
184/// let due = client.get_due_cards("Spanish::Vocab").await?;
185/// ```
186pub struct AnkiConnectClient {
187    /// HTTP client for JSON-RPC requests.
188    http: reqwest::Client,
189    /// Base URL of the AnkiConnect endpoint.
190    base_url: String,
191    /// Optional API key appended as `?key=...`.
192    api_key: Option<String>,
193    /// Serializes concurrent calls — AnkiConnect is single-threaded.
194    mutex: tokio::sync::Mutex<()>,
195    /// Approved action names only.
196    allowed_actions: HashSet<&'static str>,
197}
198
199impl AnkiConnectClient {
200    /// Construct a new client targeting `localhost:{port}`.
201    ///
202    /// If `api_key` is provided it is appended as `?key={api_key}` to every
203    /// request, matching AnkiConnect's optional authentication mechanism.
204    pub fn new(port: u16, api_key: Option<String>) -> Self {
205        let mut allowed = HashSet::new();
206
207        // Read-only — always permitted
208        allowed.insert("deckNames");
209        allowed.insert("getDeckStats");
210        allowed.insert("findCards");
211        allowed.insert("findNotes");
212        allowed.insert("cardsInfo");
213        allowed.insert("notesInfo");
214        allowed.insert("getDueCards");
215        // (getDueCards is implemented via findCards, not a native action)
216        allowed.insert("version");
217
218        // Sync — requires Anki Web auth configured in Anki
219        allowed.insert("sync");
220
221        // Safe write operations — scheduler-aware
222        allowed.insert("suspend");
223        allowed.insert("unsuspend");
224        allowed.insert("setDueDate");
225        // Note: exportPackage used for backups (no native createBackup action)
226        allowed.insert("exportPackage");
227        allowed.insert("importPackage");
228
229        let base_url = format!("http://localhost:{port}");
230        let http = reqwest::Client::builder()
231            .timeout(std::time::Duration::from_secs(30))
232            .build()
233            .expect("reqwest::Client::builder should not fail");
234
235        Self {
236            http,
237            base_url,
238            api_key,
239            mutex: tokio::sync::Mutex::new(()),
240            allowed_actions: allowed,
241        }
242    }
243
244    /// Verify AnkiConnect is reachable and reports API version 6.
245    ///
246    /// Calls the `version` action internally. Returns `Ok(())` if Anki is
247    /// running and responding with version 6. Returns `Err` otherwise.
248    pub async fn health_check(&self) -> Result<(), AnkiConnectError> {
249        let raw = self.invoke_raw("version", serde_json::json!({})).await?;
250        let version: i64 = serde_json::from_value(raw)
251            .map_err(AnkiConnectError::JsonError)?;
252        if version != 6 {
253            return Err(AnkiConnectError::VersionMismatch(version));
254        }
255        Ok(())
256    }
257
258    /// Low-level invoke — constructs JSON-RPC payload and sends HTTP POST.
259    ///
260    /// Calls are serialized through an internal mutex. Whitelist is checked
261    /// before the request is sent. On HTTP connection failure, returns
262    /// `AnkiConnectError::Unreachable`.
263    ///
264    /// Returns the raw JSON value from the `result` field.
265    async fn invoke_raw(&self, action: &str, params: serde_json::Value) -> Result<serde_json::Value, AnkiConnectError> {
266        if !self.allowed_actions.contains(action) {
267            return Err(AnkiConnectError::ActionNotAllowed(action.to_string()));
268        }
269
270        let url = match &self.api_key {
271            Some(key) => format!("{}?key={}", self.base_url, key),
272            None => self.base_url.clone(),
273        };
274
275        // AnkiConnect requires `params` to be a JSON object, not null.
276        // Unit structs serialize to null — normalise to empty object.
277        let params = if params.is_null() {
278            serde_json::json!({})
279        } else {
280            params
281        };
282
283        let payload = serde_json::json!({
284            "action": action,
285            "version": 6,
286            "params": params
287        });
288
289        let resp = self.http
290            .post(&url)
291            .json(&payload)
292            .send()
293            .await
294            .map_err(|e| {
295                if e.is_connect() || e.is_timeout() {
296                    // Extract port from base_url for error message
297                    let port = self.base_url
298                        .strip_prefix("http://localhost:")
299                        .unwrap_or("8765");
300                    AnkiConnectError::Unreachable(port.parse().unwrap_or(8765))
301                } else {
302                    AnkiConnectError::HttpError(e)
303                }
304            })?;
305
306        let ac_resp: AnkiConnectResponse = resp.json().await?;
307        ac_resp.into_result()
308    }
309
310    /// Invoke an action with typed parameters and return a typed result.
311    async fn invoke<P: serde::Serialize + std::fmt::Debug, R: serde::de::DeserializeOwned>(
312        &self,
313        action: &str,
314        params: P,
315    ) -> Result<R, AnkiConnectError> {
316        let raw = self.invoke_raw(action, serde_json::to_value(params)?).await?;
317        serde_json::from_value(raw).map_err(AnkiConnectError::JsonError)
318    }

337    /// Return IDs of all cards due in a named deck.
338    ///
339    /// Uses `findCards` with an `is:due` filter scoped to the deck.
340    /// This reflects the live scheduler queue — available only when Anki is
341    /// running. Cards in states: new (type=0), learning (type=1), review
342    /// (type=2, due <= now), relearning.
343    pub async fn get_due_cards(&self, deck: &str) -> Result<Vec<i64>, AnkiConnectError> {
344        let query = format!("deck:\"{}\" is:due", deck);
345        self.invoke("findCards", FindCardsParams { query }).await
346    }

352    pub async fn find_cards(&self, query: &str) -> Result<Vec<i64>, AnkiConnectError> {
353        self.invoke("findCards", FindCardsParams { query: query.to_string() }).await

375    /// Trigger a full Anki Web sync.
376    ///
377    /// Requires Anki to have sync auth configured (ankiweb.net account or
378    /// self-hosted sync server). This is the only safe way to synchronize
379    /// the collection — doing it via direct SQLite writes would corrupt the
380    /// sync log.
381    pub async fn sync(&self) -> Result<serde_json::Value, AnkiConnectError> {
382        // Hold the mutex for the full sync transaction
383        let _guard = self.mutex.lock().await;
384        self.invoke("sync", SyncParams).await
385    }

387    /// Suspend one or more cards by ID.

391    pub async fn suspend_cards(&self, cards: Vec<i64>) -> Result<bool, AnkiConnectError> {
392        self.invoke("suspend", SuspendCardsParams { cards }).await
393    }

445    /// Create a timestamped `.apkg` backup of a deck via `exportPackage`.
446    ///
447    /// AnkiConnect has no native "backup" action — `exportPackage` with
448    /// `includeSched: true` is the safe equivalent: it produces a portable
449    /// .apkg with full scheduling history. The caller provides the deck
450    /// name and a directory path; the file is named with a timestamp.
451    pub async fn create_backup(&self, deck: &str, dir: &str) -> Result<String, AnkiConnectError> {
452        let ts = chrono::Utc::now().format("%Y%m%dT%H%M%S");
453        let path = format!("{dir}/{deck}_{ts}.apkg");
454        self.export_package(deck, &path, true).await?;
455        Ok(path)
456    }
