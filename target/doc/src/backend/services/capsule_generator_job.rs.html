1//! Sequential Background Capsule Generation runner.
2//!
3//! Mitigates thundering-herd issues on concurrent SQLite reads by sequentially
4//! iterating over students, resolving their profiles, FSRS sorting their cards,
5//! and handling the anomaly fallbacks cleanly without crashing.

20// Struct to represent individual student failures (Anomalies)
21#[derive(Debug, serde::Serialize)]
22pub struct StudentAnomaly {
23    pub student_id: String,
24    pub name: String,
25    pub anomaly_type: String, // "insufficient_content", "retention_cliff", "dropout_stale"
26    pub msg: String,
27    pub fallback_applied: String,
28}

30/// Triggers and executes sequential capsule generation for all active students.
31pub async fn run_batch_generation(
32    db: &DatabaseConnection,
33    job_id: &str,
34    playground_dir: &str,
35) -> Result<(), sea_orm::DbErr> {
36    // 1. Mark job as "processing"
37    let mut job: gj_entity::ActiveModel = gj_entity::Entity::find_by_id(job_id.to_string())
38        .one(db)
39        .await?
40        .unwrap()
41        .into();
42
43    job.status = ActiveValue::Set("processing".to_string());
44    let _ = job.update(db).await?;
45
46    // 2. Fetch all active students (role = "student")
47    let students = user_entity::Entity::find()
48        .filter(user_entity::Column::Role.eq("student"))
49        .all(db)
50        .await?;
51
52    let total_students = students.len() as i32;
53
54    // Update total students in job record
55    let mut job: gj_entity::ActiveModel = gj_entity::Entity::find_by_id(job_id.to_string())
56        .one(db)
57        .await?
58        .unwrap()
59        .into();
60    job.total_students = ActiveValue::Set(total_students);
61    let _ = job.update(db).await?;
62
63    let mut completed_count = 0;
64    let mut anomalies_list: Vec<StudentAnomaly> = Vec::new();
65
66    // 3. Sequentially process each student
67    for student in &students {
68        let student_str_id = student.id.to_string();
69        let student_fullname = student.full_name.clone().unwrap_or_else(|| student.username.clone());
70
71        // Locate their .anki2 database inside the playground sandbox
72        let anki2_path = format!("{}/{}/collection.anki2", playground_dir, student_fullname);
73
74        if !Path::new(&anki2_path).exists() {
75            // Log missing file as an anomaly and skip to next student
76            anomalies_list.push(StudentAnomaly {
77                student_id: student_str_id.clone(),
78                name: student_fullname.clone(),
79                anomaly_type: "insufficient_content".to_string(),
80                msg: format!("Anki collection.anki2 file not found at: {}", anki2_path),
81                fallback_applied: "Skipped generation for this student".to_string(),
82            });
83
84            completed_count += 1;
85            update_job_progress(db, job_id, completed_count, total_students, &anomalies_list).await?;
86            continue;
87        }
88
89        // Format ISO week for compilation target
90        let iso_week = "2026-W28".to_string(); // Seed Week
91
92        // Trigger safe gen capsule request
93        let request = GenerateRequest {
94            student_id: student_str_id.clone(),
95            anki2_path: anki2_path.clone(),
96            track_profile_id: String::new(),
97            session_week: iso_week,
98            n_value: None,
99            session_duration_minutes: None,
100        };
101
102        // Standard Generator invocation
103        match capsule_generator::generate_capsule(db, request).await {
104            Ok(_response) => {
105                // Success! Record it.
106            }
107            Err(GeneratorError::RateLimited(_)) => {
108                // Rate limited in this window — skip normally
109            }
110            Err(e) => {
111                // Determine anomaly categorization
112                let err_str = e.to_string();
113                let anomaly_type = if err_str.contains("NoDueCards") || err_str.contains("NoTracks") {
114                    "insufficient_content".to_string()
115                } else if err_str.contains("rate limited") {
116                    "rate_limited".to_string()
117                } else {
118                    "retention_cliff".to_string()
119                };
120
121                // Store error details
122                anomalies_list.push(StudentAnomaly {
123                    student_id: student_str_id.clone(),
124                    name: student_fullname.clone(),
125                    anomaly_type,
126                    msg: err_str,
127                    fallback_applied: "NoDueCards fallback applied (deck defaulted)".to_string(),
128                });
129            }
130        }
131
132        completed_count += 1;
133        update_job_progress(db, job_id, completed_count, total_students, &anomalies_list).await?;
134    }
135
136    // 4. Mark job as "completed" or "failed"
137    let mut job: gj_entity::ActiveModel = gj_entity::Entity::find_by_id(job_id.to_string())
138        .one(db)
139        .await?
140        .unwrap()
141        .into();
142
143    job.status = ActiveValue::Set("completed".to_string());
144    job.progress = ActiveValue::Set(100);
145    job.completed_at = ActiveValue::Set(Some(chrono::Utc::now().timestamp()));
146    let _ = job.update(db).await?;
147
148    Ok(())
149}

151async fn update_job_progress(
152    db: &DatabaseConnection,
153    job_id: &str,
154    completed: i32,
155    total: i32,
156    anomalies: &Vec<StudentAnomaly>,
157) -> Result<(), sea_orm::DbErr> {
158    let mut job: gj_entity::ActiveModel = gj_entity::Entity::find_by_id(job_id.to_string())
159        .one(db)
160        .await?
161        .unwrap()
162        .into();
163
164    let prog_pct = if total > 0 { (completed * 100) / total } else { 100 };
165
166    job.progress = ActiveValue::Set(prog_pct);
167    job.completed_students = ActiveValue::Set(completed);
168
169    if !anomalies.is_empty() {
170        let json_anoms = serde_json::to_string(anomalies).unwrap_or_else(|_| "[]".to_string());
171        job.anomalies = ActiveValue::Set(Some(json_anoms));
172    }
173
174    let _ = job.update(db).await?;
175    Ok(())
176}
