1//! Capsule Delivery — HTTP client for the AnkiTov Session Driver addon.
2//!
3//! After the backend generates a remediation capsule (card IDs), this client
4//! delivers it to the Anki addon running inside the student's Anki container.
5//! The addon (port 18765) creates a filtered deck, tags the capsule cards,
6//! and opens Anki's reviewer so the student can begin their session.
7//!
8//! ## Protocol
9//!
10//! | Method | Path | Purpose |
11//! |--------|------|---------|
12//! | POST | `/session` | Start a new capsule review session |
13//! | GET | `/status` | Check current session state |
14//! | DELETE | `/session` | Cancel the active session |
15//!
16//! ## Configuration
17//!
18//! | Env var | Default | Description |
19//! |---------|---------|-------------|
20//! | `ANKITOV_CAPSULE_PORT` | `18765` | Addon HTTP listener port |
21
22use reqwest::Client;
23use serde::{Deserialize, Serialize};
24use std::time::Duration;
25
26/// Default port for the session driver addon.
27const DEFAULT_CAPSULE_PORT: u16 = 18765;
28
29/// HTTP timeout for addon communication.
30const REQUEST_TIMEOUT: Duration = Duration::from_secs(10);
31
32// ---------------------------------------------------------------------------
33// Request / Response types
34// ---------------------------------------------------------------------------
35
36/// Payload sent to the addon to start a session.
37#[derive(Debug, Serialize)]
38pub struct StartSessionRequest {
39    pub session_uuid: String,
40    pub card_ids: Vec<i64>,
41    pub capsule_size: usize,
42    pub track_profile_name: String,
43}
44
45/// Response from the addon after a `/session` POST.
46#[derive(Debug, Serialize, Deserialize)]
47pub struct StartSessionResponse {
48    pub status: String,
49    pub session_uuid: String,
50    #[serde(default)]
51    pub deck_name: Option<String>,
52    #[serde(default)]
53    pub card_count: usize,
54    #[serde(default)]
55    pub replaced_session: Option<String>,
56}
57
58/// Response from the addon after a `/status` GET.
59#[derive(Debug, Deserialize)]
60pub struct SessionStatus {
61    pub session_active: bool,
62    #[serde(default)]
63    pub session_uuid: Option<String>,
64    #[serde(default)]
65    pub state: Option<String>,
66    #[serde(default)]
67    pub cards_total: usize,
68    #[serde(default)]
69    pub cards_reviewed: usize,
70}
71
72// ---------------------------------------------------------------------------
73// Errors
74// ---------------------------------------------------------------------------
75
76#[derive(Debug, thiserror::Error)]
77pub enum DeliveryError {
78    #[error("addon unreachable at {url}: {source}")]
79    Unreachable {
80        url: String,
81        source: reqwest::Error,
82    },
83
84    #[error("addon returned error: {0}")]
85    AddonError(String),
86
87    #[error("HTTP error: {0}")]
88    Http(#[from] reqwest::Error),
89
90    #[error("JSON error: {0}")]
91    Json(#[from] serde_json::Error),
92
93    #[error("addon returned unexpected status: {0}")]
94    UnexpectedStatus(String),
95}
96
97// ---------------------------------------------------------------------------
98// Client
99// ---------------------------------------------------------------------------
100
101/// HTTP client for the AnkiTov Session Driver addon.
102///
103/// ```rust,ignore
104/// let client = CapsuleDelivery::new(18765);
105/// let response = client.start_session(
106///     "uuid-1234",
107///     &[1001, 1002, 1003],
108///     25,
109///     "Grade 7 Math Profile",
110/// ).await?;
111/// ```
112pub struct CapsuleDelivery {
113    http: Client,
114    base_url: String,
115}
116
117impl CapsuleDelivery {
118    /// Create a new delivery client targeting the addon on the given port.
119    pub fn new(port: u16) -> Self {
120        let base_url = format!("http://127.0.0.1:{port}");
121        let http = Client::builder()
122            .timeout(REQUEST_TIMEOUT)
123            .build()
124            .expect("reqwest::Client::builder should not fail");
125
126        Self { http, base_url }
127    }
128
129    /// Resolve the addon port from environment, or use the default (18765).
130    pub fn from_env() -> Self {
131        let port = std::env::var("ANKITOV_CAPSULE_PORT")
132            .ok()
133            .and_then(|s| s.parse().ok())
134            .unwrap_or(DEFAULT_CAPSULE_PORT);
135        Self::new(port)
136    }
137
138    /// Deliver a capsule to the addon and start a review session.
139    ///
140    /// Sends `POST /session` with the capsule metadata. The addon will:
141    /// 1. Tag all card_ids with `ankitov-capsule-<session_uuid>`
142    /// 2. Create a filtered deck "AnkiTov Capsule" with those cards
143    /// 3. Open the reviewer in random order (interleaved)
144    ///
145    /// Returns the addon's response, which includes the session status.
146    pub async fn start_session(
147        &self,
148        session_uuid: &str,
149        card_ids: &[i64],
150        capsule_size: usize,
151        track_profile_name: &str,
152    ) -> Result<StartSessionResponse, DeliveryError> {
153        let url = format!("{}/session", self.base_url);
154
155        let payload = StartSessionRequest {
156            session_uuid: session_uuid.to_string(),
157            card_ids: card_ids.to_vec(),
158            capsule_size,
159            track_profile_name: track_profile_name.to_string(),
160        };
161
162        let resp = self
163            .http
164            .post(&url)
165            .json(&payload)
166            .send()
167            .await
168            .map_err(|e| {
169                if e.is_connect() || e.is_timeout() {
170                    DeliveryError::Unreachable {
171                        url: url.clone(),
172                        source: e,
173                    }
174                } else {
175                    DeliveryError::Http(e)
176                }
177            })?;
178
179        if !resp.status().is_success() {
180            let status_code = resp.status().as_u16();
181            let body = resp.text().await.unwrap_or_default();
182            return Err(DeliveryError::AddonError(format!(
183                "HTTP {status_code}: {body}"
184            )));
185        }
186
187        let response: StartSessionResponse = resp.json().await?;
188
189        match response.status.as_str() {
190            "started" | "replaced" | "empty" => Ok(response),
191            "error" => Err(DeliveryError::AddonError(
192                format!(
193                    "addon rejected session {}: {:?}",
194                    response.session_uuid,
195                    response.status
196                )
197            )),
198            other => Err(DeliveryError::UnexpectedStatus(other.to_string())),
199        }
200    }
201
202    /// Query the addon for the current session status.
203    ///
204    /// Returns information about the active session: how many cards have been
205    /// reviewed, the current state (IDLE / TAGGING / REVIEWING / CLEANUP), and
206    /// the session UUID if one is active.
207    pub async fn get_status(&self) -> Result<SessionStatus, DeliveryError> {
208        let url = format!("{}/status", self.base_url);
209
210        let resp = self
211            .http
212            .get(&url)
213            .send()
214            .await
215            .map_err(|e| {
216                if e.is_connect() || e.is_timeout() {
217                    DeliveryError::Unreachable {
218                        url: url.clone(),
219                        source: e,
220                    }
221                } else {
222                    DeliveryError::Http(e)
223                }
224            })?;
225
226        let status: SessionStatus = resp.json().await?;
227        Ok(status)
228    }
229
230    /// Cancel the active session on the addon.
231    ///
232    /// Sends `DELETE /session`. The addon will remove the filtered deck,
233    /// untag the cards, and reset to IDLE state.
234    pub async fn cancel_session(&self) -> Result<serde_json::Value, DeliveryError> {
235        let url = format!("{}/session", self.base_url);
236
237        let resp = self
238            .http
239            .delete(&url)
240            .send()
241            .await
242            .map_err(|e| {
243                if e.is_connect() || e.is_timeout() {
244                    DeliveryError::Unreachable {
245                        url: url.clone(),
246                        source: e,
247                    }
248                } else {
249                    DeliveryError::Http(e)
250                }
251            })?;
252
253        let body: serde_json::Value = resp.json().await?;
254        Ok(body)
255    }
256
257    /// Check if the addon is reachable (health check).
258    ///
259    /// Returns `true` if the addon responds to a status query.
260    pub async fn health_check(&self) -> bool {
261        self.get_status().await.is_ok()
262    }
263}
