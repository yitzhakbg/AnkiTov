1//! Capsule Generator — server-side pipeline that produces a single interleaved
2//! "Remediation Capsule" per student from a customized cocktail of prerequisite
3//! tracks.
4//!
5//! ## Pipeline Flow
6//!
7//! ```text
8//! resolve profile ──→ query due cards ──→ FSRS sort ──→ slice ──→ persist
9//! ```
10//!
11//! 1. Resolve the student's TrackProfile (N, session_duration, track tags) via SeaORM.
12//! 2. Query the `.anki2` SQLite database for all overdue cards matching assigned
13//!    track tags (`fsrs_sort::fetch_and_sort`).
14//! 3. Slice the ranked list to the computed capsule size (`capsule_slicer`).
15//! 4. Persist a `CapsuleSession` entity with status `pending`.
16//! 5. Return the capsule card IDs so the PWA can deliver them to Anki.
17//!
18//! ## Configuration
19//!
20//! The generator reads from:
21//! - SeaORM (libSQL): track profiles, tracks, capsule sessions
22//! - SQLite (.anki2): card data, revlog timing history (via `fsrs_sort`)
23//!
24//! All writes go through SeaORM. The `.anki2` file is only read, never written.

47#[derive(Debug, thiserror::Error)]
48pub enum GeneratorError {
49    #[error("database error: {0}")]
50    Database(#[from] sea_orm::DbErr),
51
52    #[error("fsrs sort error: {0}")]
53    FsrsSort(#[from] fsrs_sort::FsrsSortError),
54
55    #[error("no track profile assigned for student: {0}")]
56    NoProfileForStudent(String),
57
58    #[error("no tracks assigned to profile: {0}")]
59    NoTracksInProfile(String),
60
61    #[error("profile not found: {0}")]
62    ProfileNotFound(String),
63
64    #[error("no due cards found for the assigned tracks")]
65    NoDueCards,
66
67    #[error("rate limited: student {0} already had a session in the last 30 minutes")]
68    RateLimited(String),
69
70    #[error("serialization error: {0}")]
71    Serialize(#[from] serde_json::Error),
72}

78/// Input to the capsule generator.

116/// Resolves the correct TrackProfile ID for a student using the Hybrid lookup:
117/// 1. Student Override (target_type = "student", target_id = student_id)
118/// 2. Cohort Default (target_type = "cohort", target_id = user.subsection_id)
119/// 3. Fallback (First available TrackProfile)
120pub async fn resolve_profile_for_student(
121    db: &DatabaseConnection,
122    student_id: &str,
123) -> Result<String, GeneratorError> {
124    // 1. Check Student Override
125    let student_override = pa_entity::Entity::find()
126        .filter(pa_entity::Column::TargetType.eq("student"))
127        .filter(pa_entity::Column::TargetId.eq(student_id))
128        .one(db)
129        .await
130        .map_err(GeneratorError::Database)?;
131
132    if let Some(assignment) = student_override {
133        return Ok(assignment.track_profile_id);
134    }
135
136    // 2. Check Cohort Default (requires fetching student's user record)
137    let id_val = student_id.parse::<i64>().unwrap_or(0);
138    let student_user = user_entity::Entity::find_by_id(id_val)
139        .one(db)
140        .await
141        .map_err(GeneratorError::Database)?;
142
143    if let Some(user) = student_user {
144        if let Some(cohort_id) = user.subsection_id {
145            let cohort_default = pa_entity::Entity::find()
146                .filter(pa_entity::Column::TargetType.eq("cohort"))
147                .filter(pa_entity::Column::TargetId.eq(&cohort_id))
148                .one(db)
149                .await
150                .map_err(GeneratorError::Database)?;
151
152            if let Some(assignment) = cohort_default {
153                return Ok(assignment.track_profile_id);
154            }
155        }
156    }
157
158    // 3. Fallback: Legacy "First Profile Wins"
159    let junctions = tpt_entity::Entity::find()
160        .all(db)
161        .await
162        .map_err(GeneratorError::Database)?;
163
164    if let Some(j) = junctions.first() {
165        return Ok(j.track_profile_id.clone());
166    }
167
168    Err(GeneratorError::NoProfileForStudent(student_id.to_string()))
169}

175/// Generate a remediation capsule for a student.
176///
177/// This is the main entry point for the Interleaved Mastery Pipeline (Phase 3).
178/// It resolves the student's assigned track profile, queries overdue cards from
179/// their `.anki2` database, ranks them by FSRS urgency, slices to the optimal
180/// capsule size, and persists a `CapsuleSession` record.
181///
182/// # Rate Limiting
183///
184/// Enforces a 30-minute cooldown: at most 1 capsule per student per
185/// 30-minute window. Returns [`GeneratorError::RateLimited`] if a
186/// `pending` or `in_progress` session exists within that window.
187pub async fn generate_capsule(
188    db: &DatabaseConnection,
189    request: GenerateRequest,
190) -> Result<GenerateResponse, GeneratorError> {
191    let student_id = &request.student_id;
192    let session_week = if request.session_week.is_empty() {
193        &current_iso_week()
194    } else {
195        &request.session_week
196    };
197
198    // ── Step 0: Rate-limit guard (30-minute window) ──────────────────
199    // Max 1 capsule per student per 30-minute window per the plan spec.
200    let thirty_min_ago = now_ts() - 1800;
201    let existing = cs_entity::Entity::find()
202        .filter(cs_entity::Column::StudentId.eq(student_id))
203        .filter(cs_entity::Column::CreatedAt.gte(thirty_min_ago))
204        .filter(
205            cs_entity::Column::Status
206                .eq("pending")
207                .or(cs_entity::Column::Status.eq("in_progress")),
208        )
209        .count(db)
210        .await
211        .map_err(GeneratorError::Database)?;
212
213    if existing > 0 {
214        return Err(GeneratorError::RateLimited(student_id.clone()));
215    }
216
217    // ── Step 1: Resolve TrackProfile ──────────────────────────────────
218    // Direct profile ID takes precedence over student assignment lookup
219    let profile_id = if !request.track_profile_id.is_empty() {
220        request.track_profile_id.clone()
221    } else {
222        resolve_profile_for_student(db, student_id).await?
223    };
224
225    let profile = tp_entity::Entity::find_by_id(&profile_id)
226        .one(db)
227        .await
228        .map_err(GeneratorError::Database)?
229        .ok_or_else(|| GeneratorError::ProfileNotFound(profile_id.clone()))?;
230
231    // ── Step 2: Get assigned track tags ──────────────────────────────
232    let track_junctions = tpt_entity::Entity::find()
233        .filter(tpt_entity::Column::TrackProfileId.eq(&profile_id))
234        .all(db)
235        .await
236        .map_err(GeneratorError::Database)?;
237
238    if track_junctions.is_empty() {
239        return Err(GeneratorError::NoTracksInProfile(profile_id));
240    }
241
242    let track_ids: Vec<String> = track_junctions.iter().map(|j| j.track_id.clone()).collect();
243    let tracks = track_entity::Entity::find()
244        .filter(track_entity::Column::Id.is_in(track_ids))
245        .all(db)
246        .await
247        .map_err(GeneratorError::Database)?;
248
249    let track_tags: Vec<String> = tracks.iter().map(|t| t.tag.clone()).collect();
250
251    // ── Step 3: Query & sort due cards via FSRS ──────────────────────
252    let anki2_path = Path::new(&request.anki2_path);
253    let sort_result = fsrs_sort::fetch_and_sort(anki2_path, &track_tags, 30)?;
254
255    if sort_result.ranked.is_empty() {
256        return Err(GeneratorError::NoDueCards);
257    }
258
259    // ── Step 4: Compute capsule size ─────────────────────────────────
260    let n_value = request.n_value
261        .or(profile.n_value)
262        .unwrap_or(capsule_slicer::DEFAULT_N_VALUE);
263    let session_duration = match request.session_duration_minutes {
264        Some(dur) if dur > 0 => dur,
265        _ => {
266            if profile.session_duration_minutes > 0 {
267                profile.session_duration_minutes
268            } else {
269                capsule_slicer::DEFAULT_SESSION_DURATION_MINUTES
270            }
271        }
272    };
273
274    let pool_size = sort_result.ranked.len();
275    let num_tracks = sort_result.track_counts.len();
276    let avg_seconds = sort_result.avg_seconds_per_card;
277
278    let target_size = capsule_slicer::compute_capsule_size(
279        pool_size,
280        n_value,
281        capsule_slicer::DEFAULT_HARD_CAP,
282        session_duration,
283        avg_seconds,
284        num_tracks,
285    );
286
287    // ── Step 5: Slice ────────────────────────────────────────────────
288    let ranked_pairs: Vec<(i64, String)> = sort_result
289        .ranked
290        .iter()
291        .map(|c| (c.card_id, c.track_tag.clone()))
292        .collect();
293
294    let slice = capsule_slicer::slice_capsule(
295        &ranked_pairs,
296        &sort_result.track_counts,
297        target_size,
298    );
299
300    // ── Step 6: Persist CapsuleSession ───────────────────────────────
301    let session_id = uuid::Uuid::new_v4().to_string();
302    let card_ids_json = serde_json::to_string(&slice.selected)?;
303    let ts = now_ts();
304
305    let session = cs_entity::ActiveModel {
306        id: ActiveValue::Set(session_id.clone()),
307        student_id: ActiveValue::Set(student_id.clone()),
308        track_profile_id: ActiveValue::Set(profile_id.clone()),
309        n_value: ActiveValue::Set(n_value),
310        session_duration_minutes: ActiveValue::Set(session_duration),
311        card_ids: ActiveValue::Set(card_ids_json),
312        status: ActiveValue::Set("pending".to_string()),
313        session_week: ActiveValue::Set(session_week.clone()),
314        cards_completed: ActiveValue::Set(0),
315        cards_total: ActiveValue::Set(slice.selected.len() as i32),
316        created_at: ActiveValue::Set(ts),
317        completed_at: ActiveValue::Set(None),
318    }
319    .insert(db)
320    .await
321    .map_err(GeneratorError::Database)?;
322
323    // Audit: log capsule generation
324    audit_logger::log_capsule_generate(
325        db,
326        student_id,
327        &profile_id,
328        session.cards_total,
329        n_value,
330        &session.id,
331    )
332    .await;
333
334    tracing::info!(
335        "Capsule generated: session={}, student={}, cards={}, N={}, duration={}min",
336        session.id,
337        student_id,
338        session.cards_total,
339        n_value,
340        session_duration,
341    );
342
343    Ok(GenerateResponse {
344        session_id: session.id.clone(),
345        card_ids: slice.selected,
346        capsule_size: slice.actual_size,
347        n_value,
348        session_duration_minutes: session_duration,
349        status: session.status.clone(),
350    })
351}

353/// Compute the current ISO week string (e.g., "2026-W27").
354pub fn current_iso_week() -> String {
355    use chrono::Datelike;
356    let now = chrono::Utc::now();
357    let iso = now.date_naive().iso_week();
358    format!("{}-W{:02}", iso.year(), iso.week())
359}

365fn now_ts() -> i64 {
366    std::time::SystemTime::now()
367        .duration_since(std::time::UNIX_EPOCH)
368        .unwrap()
369        .as_secs() as i64
370}
