1//! Telemetry Ingestor — consume card-answer events into the audit ledger.
2//!
3//! Receives telemetry pings from the DRM addon (via the backend API) and
4//! writes them to the immutable append-only audit_log table. Also updates
5//! per-track FSRS health metrics for the Management Console dashboard.
6//!
7//! The DRM addon fires on `reviewer_did_answer_card` and POSTs:
8//! ```json
9//! {
10//!     "student_id": "...",
11//!     "card_id": 123456,
12//!     "deck_name": "Math Grade 7",
13//!     "ease": 3,
14//!     "time_ms": 4521,
15//!     "session_uuid": "..."  // from cross-addon contract (v2)
16//! }
17//! ```
18//!
19//! v1: The backend resolves the card's tags via AnkiConnect to determine
20//!     which track(s) the card belongs to. The session is inferred from
21//!     the active capsule tag.
22//!
23//! v2: The session driver sets `mw.ankitov_session_uuid` for the DRM addon
24//!     to include directly.
25

35/// Incoming telemetry ping from the DRM addon.
36#[derive(Debug, Deserialize)]
37pub struct CardAnswerEvent {
38    pub student_id: String,
39    pub card_id: i64,
40    pub deck_name: String,
41    /// Ease: 1=Again, 2=Hard, 3=Good, 4=Easy
42    pub ease: i32,
43    /// Time spent on the card in milliseconds.
44    pub time_ms: i64,
45    /// Optional session UUID (v2 cross-addon contract).
46    #[serde(default)]
47    pub session_uuid: Option<String>,
48}
49
50/// Track-level telemetry snapshot for the dashboard.
51#[derive(Debug, Clone, serde::Serialize)]
52pub struct TrackTelemetry {
53    pub track_tag: String,
54    pub total_answers: u64,
55    pub avg_ease: f64,
56    pub avg_time_secs: f64,
57    /// Approximate stability trend: avg(ivl) from the last 7 days.
58    pub avg_interval_days: f64,
59}

65/// Ingest a card-answer telemetry event.
66///
67/// Records the event in the audit ledger and returns a track tag
68/// lookup key for the caller to update per-track health metrics.
69pub async fn ingest_card_answer(
70    db: &DatabaseConnection,
71    event: &CardAnswerEvent,
72) -> Result<(), TelemetryError> {
73    let event_data = serde_json::json!({
74        "student_id": event.student_id,
75        "card_id": event.card_id,
76        "deck_name": event.deck_name,
77        "ease": event.ease,
78        "time_ms": event.time_ms,
79        "session_uuid": event.session_uuid,
80        "ingested_at": now_ts(),
81    });
82
83    audit_logger::log_event(
84        db,
85        "card_answer",
86        &event_data.to_string(),
87        &event.student_id,
88        None,
89    )
90    .await;
91
92    tracing::debug!(
93        "Telemetry ingested: student={} card={} ease={} time={}ms",
94        event.student_id,
95        event.card_id,
96        event.ease,
97        event.time_ms,
98    );
99
100    Ok(())
101}
102
103/// Ingest a batch of card-answer events efficiently.
104///
105/// In production, the DRM addon may buffer answers and send them in
106/// batches every few seconds to reduce HTTP overhead.
107pub async fn ingest_batch(
108    db: &DatabaseConnection,
109    events: &[CardAnswerEvent],
110) -> Result<usize, TelemetryError> {
111    let count = events.len();
112    for event in events {
113        ingest_card_answer(db, event).await?;
114    }
115    tracing::info!("Batch ingested: {} card-answer events", count);
116    Ok(count)
117}

129// ---------------------------------------------------------------------------

133fn now_ts() -> i64 {
134    std::time::SystemTime::now()
135        .duration_since(std::time::UNIX_EPOCH)
136        .unwrap()
137        .as_secs() as i64
138}

144#[cfg(test)]
145mod tests {
146    use super::*;
147
148    #[test]
149    fn test_card_answer_event_deserialization() {
150        let json = r#"{
151            "student_id": "student-1",
152            "card_id": 123456,
153            "deck_name": "Math Grade 7",
154            "ease": 3,
155            "time_ms": 4521
156        }"#;
157        let event: CardAnswerEvent = serde_json::from_str(json).unwrap();
158        assert_eq!(event.student_id, "student-1");
159        assert_eq!(event.card_id, 123456);
160        assert_eq!(event.ease, 3);
161        assert_eq!(event.time_ms, 4521);
162        assert!(event.session_uuid.is_none());
163    }
164
165    #[test]
166    fn test_card_answer_event_with_session() {
167        let json = r#"{
168            "student_id": "student-1",
169            "card_id": 789,
170            "deck_name": "Vocab",
171            "ease": 4,
172            "time_ms": 2100,
173            "session_uuid": "abc-123"
174        }"#;
175        let event: CardAnswerEvent = serde_json::from_str(json).unwrap();
176        assert_eq!(event.session_uuid.as_deref(), Some("abc-123"));
177    }
