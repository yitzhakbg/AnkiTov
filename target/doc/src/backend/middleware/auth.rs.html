1//! Auth middleware — JWT extraction and user injection.
2//!
3//! Extracts the `Authorization: Bearer <token>` header, decodes the JWT,
4//! and injects `AuthUser` into request extensions for downstream handlers.
5//!
6//! ## Usage
7//!
8//! ```rust,ignore
9//! use crate::middleware::auth::{require_auth, AuthUser};
10//!
11//! async fn protected_route(
12//!     Extension(user): Extension<AuthUser>,
13//! ) -> Result<Json<...>> {
14//!     // user.id, user.role, user.email are available
15//! }
16//! ```
17

29/// Authenticated user information injected into request extensions.
30#[derive(Debug, Clone, Serialize, Deserialize)]
31pub struct AuthUser {

33    pub role: String,
34    pub email: String,
35}

38/// Build a 401 `Response` without ambiguity over which `StatusCode` type is
39/// in scope (Loco's prelude pulls in `reqwest::StatusCode`, which shadows the
40/// `http` crate's one). Constructs the response directly from the axum `http`
41/// crate type.
42fn unauthorized_response() -> Response {
43    use axum::body::Body;
44    use axum::http::{self, Response as _};
45    let mut res: axum::http::Response<Body> = http::Response::builder()
46        .status(401)
47        .body(Body::empty())
48        .expect("building a 401 response");
49    res
50}

52/// Axum middleware: extract JWT from Authorization header, inject AuthUser.
53///
54/// Returns 401 if the header is missing, the token is expired, or the
55/// signature is invalid.
56///
57/// **Important:** this returns a real `Response` for the 401 (not
58/// `Err(StatusCode)`). A middleware whose error type is a `StatusCode` has
59/// its rejection silently dropped in Loco's layer chain (the error is never
60/// converted via `IntoResponse`), so the real handler would run
61/// unauthenticated. Returning `Response` directly makes the 401 bullet-proof.
62pub async fn require_auth(mut req: Request, next: Next) -> Result<Response, std::convert::Infallible> {
63    // Extract Bearer token
64    let auth_header = req
65        .headers()
66        .get(header::AUTHORIZATION)
67        .and_then(|v| v.to_str().ok())
68        .and_then(|v| v.strip_prefix("Bearer "));
69
70    let token = match auth_header {
71        Some(t) => t,
72        None => return Ok(unauthorized_response()),
73    };
74
75    // Decode and validate
76    let claims = match jwt::decode_token(token) {
77        Ok(c) => c,
78        Err(_) => return Ok(unauthorized_response()),
79    };
80
81    // Inject user into request extensions
82    req.extensions_mut().insert(AuthUser {
83        id: claims.sub,
84        role: claims.role,
85        email: claims.email,
86    });
87
88    Ok(next.run(req).await)
89}

91/// Middleware factory for **selective** management auth.
92///
93/// Loco's `AppRoutes::to_router` applies its default middleware stack
94/// (`limit_payload`, `cors`, `catch_panic`, `etag`, `request_id`, `fallback`,
95/// …) to the outer router via `Router::layer` *before* `.with_state`. A
96/// separate `Router::nest` for the management routes does not survive that
97/// chain — the nested routes stop matching and every `/api/v1/management/*`
98/// request falls through to Loco's 200 "Welcome" fallback, with `require_auth`
99/// never consulted.
100///
101/// The reliable pattern is therefore to register the management routes on the
102/// *outer* router (so Loco's own layer chain, which is proven to work, wraps
103/// them) and gate them with this middleware, which enforces JWT **only** for
104/// the `/api/v1/management` prefix and passes every other path through
105/// untouched. Apply it with `axum::middleware::from_fn` inside a
106/// `Router::layer` call in `after_routes`.
107pub async fn require_auth_for_management(
108    mut req: Request,
109    next: Next,
110) -> Result<Response, std::convert::Infallible> {
111    let p = req.uri().path();
112    // Gate both the staff console AND the student self-service surface.
113    // `/display/*` is intentionally NOT gated — the per-class display token is
114    // its own credential (spec §8.1).
115    let is_gated = p.starts_with("/api/v1/management")
116        || p.starts_with("/management")
117        || p.starts_with("/api/v1/student")
118        || p.starts_with("/student");
119    if !is_gated {
120        // Public path — pass through without touching the auth header.
121        return Ok(next.run(req).await);
122    }
123    require_auth(req, next).await
124}

126/// Optional auth: extracts JWT if present, but does not reject unauthenticated
127/// requests. Useful for routes that behave differently for logged-in users.
128pub async fn optional_auth(mut req: Request, next: Next) -> Result<Response, std::convert::Infallible> {
129    if let Some(token) = req
130        .headers()
131        .get(header::AUTHORIZATION)
132        .and_then(|v| v.to_str().ok())
133        .and_then(|v| v.strip_prefix("Bearer "))
134    {
135        if let Ok(claims) = jwt::decode_token(token) {
136            req.extensions_mut().insert(AuthUser {
137                id: claims.sub,
138                role: claims.role,
139                email: claims.email,
140            });
141        }
142    }
143
144    Ok(next.run(req).await)
145}

147/// Extract an `AuthUser` from request extensions (panics if missing — only
148/// call this on routes protected by `require_auth`).
149pub fn get_auth_user(req: &Request) -> &AuthUser {
150    req.extensions()
151        .get::<AuthUser>()
152        .expect("AuthUser not found in request extensions — is require_auth middleware applied?")
153}

155/// Gate a request by role.
156///
157/// Call at the top of a *write* handler (create / update / delete / enroll /
158/// distribute / generate / sync / install) after `require_auth` has injected
159/// the user:
160///
161/// ```rust,ignore
162/// let _ = ensure_role(&user, &["teacher", "admin"])?; // 403 for students
163/// ```
164///
165/// - Any role present in `allowed` (case-insensitive) passes.
166/// - A role *not* in the list (e.g. `"student"` when only teachers may write)
167///   yields a **403 Forbidden**.
168///
169/// Returns a [`loco_rs::Error::CustomError`] carrying `403`, so handlers can
170/// use `?` directly — the framework converts it to an HTTP 403 response.
171pub fn ensure_role(user: &AuthUser, allowed: &[&str]) -> Result<(), loco_rs::Error> {
172    let ok = allowed
173        .iter()
174        .any(|r| r.eq_ignore_ascii_case(&user.role));
175    if ok {
176        Ok(())
177    } else {
178        Err(loco_rs::Error::CustomError(
179            axum::http::StatusCode::FORBIDDEN,
180            loco_rs::controller::ErrorDetail {
181                error: Some("forbidden".into()),
182                description: Some("this action requires a teacher or admin role".into()),
183            },
184        ))
185    }
186}
