1//! Internationalization (i18n) module for AnkiTov.
2//!
3//! Uses `fluent-templates` to compile Fluent (.ftl) locale files into the
4//! binary at build time via the `static_loader!` macro. Language detection
5//! is done via a runtime override (set by the frontend language switcher).
6//!
7//! ## Architecture
8//!
9//! - Locale files: `backend/i18n/{locale}/main.ftl`
10//! - At build time, `static_loader!` compiles all .ftl files into the binary
11//! - `t(msg_id)` resolves messages at runtime for the current locale
12//! - Frontend sets locale preference via `POST /api/v1/locale/set`
13//!
14//! ## Adding a new language
15//!
16//! 1. Create `backend/i18n/{locale-code}/main.ftl`
17//! 2. Add the locale to the `LOCALES` static loader below
18//! 3. Add metadata to `SUPPORTED_LOCALES`
19//! 4. Rebuild
20
21use fluent_templates::{static_loader, Loader};
22use i18n_embed::unic_langid::LanguageIdentifier;
23use std::sync::RwLock;
24
25// ---------------------------------------------------------------------------
26// 1. Compile-time Fluent template loading
27// ---------------------------------------------------------------------------
28
29// Statically compiled Fluent bundles containing all locales.
30// Each locale directory under `i18n/` is compiled into its own bundle.
31// The fallback language is `en-US` — any missing messages resolve to
32// the English version.
33//
34// We disable Unicode isolating characters (Fluent inserts them by default
35// to prevent text-direction bleed) because we handle RTL via CSS `dir`.
36static_loader! {
37    static LOCALES = {
38        locales: "i18n",
39        fallback_language: "en-US",
40        customise: |b| b.set_use_isolating(false),
41    };
42}
43
44// ---------------------------------------------------------------------------
45// 2. Runtime locale override
46// ---------------------------------------------------------------------------
47
48/// A thread-safe override for the locale. When set, it takes precedence over
49/// the default (`en-US`). Set by the frontend via `POST /api/v1/locale/set`.
50static LOCALE_OVERRIDE: RwLock<Option<LanguageIdentifier>> = RwLock::new(None);
51
52/// List of supported locale codes and their metadata.
53/// (code, native_name, english_name)
54pub const SUPPORTED_LOCALES: &[(&str, &str, &str)] = &[
55    ("en-US", "English (US)", "English (US)"),
56    ("ar",    "العربية",      "Arabic"),
57    ("he",    "עברית",        "Hebrew"),
58    ("fr",    "Français",     "French"),
59    ("es",    "Español",      "Spanish"),
60    ("de",    "Deutsch",      "German"),
61    ("it",    "Italiano",     "Italian"),
62    ("zh",    "简体中文",     "Chinese (Simplified)"),
63    ("ko",    "한국어",       "Korean"),
64    ("ja",    "日本語",       "Japanese"),
65    ("ru",    "Русский",      "Russian"),
66    ("pt",    "Português (Brasil)", "Portuguese (Brazil)"),
67    ("hi",    "हिन्दी",       "Hindi"),

70/// Set the current locale override (called when user switches language).
71pub fn set_locale(locale_str: &str) {
72    if let Ok(langid) = locale_str.parse::<LanguageIdentifier>() {
73        if let Ok(mut guard) = LOCALE_OVERRIDE.write() {
74            *guard = Some(langid);
75        }
76    }
77}
78
79/// Returns the currently active locale code as a string.
80///
81/// Priority:
82/// 1. Runtime override (set by frontend language switcher)
83/// 2. Default: `en-US`
84pub fn current_locale() -> String {
85    if let Ok(guard) = LOCALE_OVERRIDE.read() {
86        if let Some(ref langid) = *guard {
87            return langid.to_string();
88        }
89    }
90    "en-US".to_string()
91}
92
93/// Returns the current LanguageIdentifier.
94fn current_langid() -> LanguageIdentifier {
95    if let Ok(guard) = LOCALE_OVERRIDE.read() {
96        if let Some(ref langid) = *guard {
97            return langid.clone();
98        }
99    }
100    "en-US".parse().unwrap()
101}

103// ---------------------------------------------------------------------------
104// 3. Public API
105// ---------------------------------------------------------------------------
106
107/// Returns true if the current locale is a Right-to-Left (RTL) language.
108pub fn is_rtl() -> bool {
109    let loc = current_locale();
110    matches!(loc.as_str(), "ar" | "he" | "fa" | "ur" | "yi" | "ku" | "sd")
111}
112
113/// Look up a translated string for the current locale.
114///
115/// Falls back to `en-US` if the message ID is not found.
116pub fn t(message_id: &str) -> String {
117    let langid = current_langid();
118    LOCALES.lookup(&langid, message_id)
119}
120
121/// Look up a translated string, returning `None` if the message requires
122/// variables that aren't provided (e.g. `{ $detail }`) or if formatting
123/// fails. Use this for flat-map endpoints that can't provide per-message
124/// args. `fluent-templates` panics on resolve errors, so we catch the panic.
125pub fn t_optional(message_id: &str) -> Option<String> {
126    let langid = current_langid();
127    // `LOCALES.try_lookup` still panics when a message references a variable
128    // that isn't provided. Catch it so flat-map endpoints degrade gracefully.
129    use std::panic::{catch_unwind, AssertUnwindSafe};
130    let result = catch_unwind(AssertUnwindSafe(|| LOCALES.try_lookup(&langid, message_id)));
131    match result {
132        Ok(Some(s)) => Some(s),
133        Ok(None) => None,
134        Err(_) => None,
135    }
136}
137
138/// Look up a translated string for a specific locale.
139pub fn t_for(locale_str: &str, message_id: &str) -> String {
140    let langid: LanguageIdentifier = locale_str.parse().unwrap_or_else(|_| "en-US".parse().unwrap());
141    LOCALES.lookup(&langid, message_id)
142}
143
144/// Look up with variable arguments.
145pub fn t_with(message_id: &str, args: &[(&str, &str)]) -> String {
146    let langid = current_langid();
147    let mut map = std::collections::HashMap::new();
148    for (k, v) in args {
149        map.insert(k.to_string(), fluent_templates::fluent_bundle::FluentValue::from(*v));
150    }
151    LOCALES.lookup_with_args(&langid, message_id, &map)
152}

162    /// Every locale directory shipped in `backend/i18n/`. Kept as an explicit
163    /// list (rather than globbed at runtime) so the test fails to *compile*

165    /// than a broken build.

182    /// A Fluent message ID is `[A-Za-z][A-Za-z0-9_-]*` — dots are NOT legal.
183    /// A dotted ID makes the parser read the ID up to the dot and then demand
184    /// `=` where it finds `.`, which aborts the whole bundle. That once shipped
185    /// 16 such IDs in `en-US` and turned 7 i18n tests red; the symptom was a
186    /// poisoned `Lazy` ("Lazy instance has previously been poisoned"), which
187    /// points at the *first* failure and hides the real cause. This test
188    /// reports the file, the byte offset and the offending line instead.
189    #[test]
190    fn every_locale_bundle_parses() {
191        let mut problems: Vec<String> = Vec::new();
192
193        for (locale, source) in BUNDLES.iter().copied() {
194            if let Err((_, errors)) = fluent_syntax::parser::parse(source) {
195                for error in errors {
196                    let pos = error.pos.start;
197                    let line_no = source[..pos].lines().count();
198                    let line = source.lines().nth(line_no.saturating_sub(1)).unwrap_or("").trim();
199                    problems.push(format!(
200                        "{locale}/main.ftl:{}: {:?} (byte {}) -> `{line}`",
201                        line_no, error.kind, pos
202                    ));
203                }
204            }
205        }
206
207        assert!(
208            problems.is_empty(),
209            "{} Fluent parse error(s). A message ID may only contain \
210             [A-Za-z0-9_-]; a '.' is illegal and breaks the entire bundle:\n  {}",
211            problems.len(),
212            problems.join("\n  ")
213        );
214    }
215
216    /// Companion to the above: assert the dotted IDs that caused the incident
217    /// are actually gone, so a future editor cannot reintroduce the shape even
218    /// if the parser ever grows to tolerate it.
219    #[test]
220    fn no_message_id_contains_a_dot() {
221        for (locale, source) in BUNDLES.iter().copied() {
222            for (i, line) in source.lines().enumerate() {
223                let trimmed = line.trim_start();
224                if trimmed.is_empty() || trimmed.starts_with('#') {
225                    continue;
226                }
227                // A message or term entry starts with `[ -]identifier =`.
228                let Some((id, tail)) = trimmed.split_once('=') else {
229                    continue;
230                };
231                if tail.starts_with('=') {
232                    continue; // a literal `==`, not an entry
233                }
234                let id = id.trim().strip_prefix('-').unwrap_or(id.trim());
235                assert!(
236                    !id.contains('.'),
237                    "{locale}/main.ftl:{}: illegal id `{id}` — dots are not legal in \
238                     Fluent identifiers (use `-`)",
239                    i + 1
240                );
241            }
242        }
243    }
