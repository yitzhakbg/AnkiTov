25/// Startup boot guard for the JWT signing secret.
26///
27/// AnkiTov issues self-contained JWTs for every authenticated request. If the
28/// hard-coded development secret is used in **production**, anyone who knows
29/// it can mint valid tokens for *any* role (including `admin`) and read or
30/// write every multi-tenant record — effectively a full authentication
31/// bypass. This guard makes that failure mode loud:
32///
33/// * **Production** (or anything not `test`/`development`) with the default
34///   secret → hard `Err`; the process refuses to boot.
35/// * **Development / test** with the default secret → `warn!` once (so local
36///   `loco start` and the test harness keep working without env setup) but the
37///   insecure default is visible in the log.
38///
39/// Call this at the very start of [`App::boot`] *before* `create_app` so a
40/// misconfigured production build never serves a single request.
41fn enforce_jwt_secret(environment: &Environment) -> Result<()> {
42    if !auth_svc::is_default_secret() {
43        // A real secret is configured — nothing to enforce.
44        return Ok(());
45    }
46
47    match environment {
48        Environment::Production => {
49            let msg = "ANKITOV_JWT_SECRET is not set (or is the hard-coded dev default).                        Refusing to boot in production — export a strong random secret,                        e.g.:  openssl rand -hex 32";
50            tracing::error!(
51                "JWT boot guard: production refuses the insecure default secret. {msg}"
52            );
53            Err(loco_rs::Error::Message(msg.to_string()))
54        }
55        Environment::Development | Environment::Test => {
56            tracing::warn!(
57                "ANKITOV_JWT_SECRET is the insecure development default. Set a real                  secret in production (e.g. `openssl rand -hex 32`). Continuing in                  `{}` mode.",
58                environment.to_string()
59            );
60            Ok(())
61        }
62        // Any other (custom / Any) environment is treated like production.
63        _ => Err(loco_rs::Error::Message(
64            "JWT boot guard: non-production environment detected with the default              ANKITOV_JWT_SECRET; refusing to start. Export a real secret.".into(),
65        )),
66    }
67}

84    fn routes(_ctx: &AppContext) -> AppRoutes {
85        // Register the full `/api/v1` table on the outer router. Loco's
86        // `to_router` flattens each controller `Routes` into top-level
87        // `Router::route()` calls and applies its default middleware chain
88        // (limit_payload, cors, catch_panic, etag, request_id, fallback, …)
89        // via `Router::layer` *before* `.with_state`.
90        //
91        // We deliberately do NOT split management out into a separate
92        // `Router::nest`: a nested router does not survive Loco's layer chain
93        // (the nested routes stop matching and every request falls through to
94        // Loco's 200 "Welcome" fallback, with `require_auth` never consulted).
95        //
96        // Instead, `after_routes` applies `require_auth_for_management` as a
97        // `Router::layer` on this outer router. That middleware enforces JWT
98        // *only* for the `/api/v1/management` prefix and passes every other
99        // path (health, auth, telemetry, dashboard, …) through untouched —
100        // so the public endpoints stay open while management is gated, all
101        // inside Loco's proven layer mechanism.
102        AppRoutes::empty()
103            .prefix("/api/v1")
104            .add_route(controllers::health::routes())
105            .add_route(controllers::auth::routes())
106            .add_route(controllers::telemetry::routes())
107            .add_route(controllers::sync::routes())
108            .add_route(controllers::management::decks::routes())
109            .add_route(controllers::management::stats::routes())
110            .add_route(controllers::management::addons::routes())
111            .add_route(controllers::management::students::routes())
112            .add_route(controllers::management::sync::routes())
113            .add_route(controllers::management::anki_ops::routes())
114            .add_route(controllers::management::probe::routes())
115            .add_route(controllers::management::jev::routes())
116            .add_route(controllers::management::tracks::routes())
117            .add_route(controllers::management::track_profiles::routes())

119            .add_route(controllers::management::classes::routes())

131    async fn after_routes(router: AxumRouter, ctx: &AppContext) -> Result<AxumRouter> {
132        use axum::Router;
133        use axum::routing::{get, post};
134        use crate::middleware::auth::require_auth_for_management;
135
136        // CORS: allow the Management Console GUI loaded from file:// or local dev.
137        let cors = CorsLayer::new()
138            .allow_origin(Any)
139            .allow_methods(Any)
140            .allow_headers(Any);
141
142        // Re-register the dashboard on the (public) outer router. It was never
143        // part of the `/api/v1` table, so it has no auth layer and stays public.
144        let dashboard = router
145            // OpenAPI spec + Scalar API reference — public, self-documenting
146            // (docs/src/reference/openapi.md). The JSON endpoint is what the
147            // Scalar page and any external tooling consume.
148            .route("/api/v1/openapi.json", get(crate::openapi::openapi_json))
149            .route("/scalar", get(crate::openapi::scalar_ui))
150            .route("/", get(controllers::dashboard::imp_console))
151            .route("/health", get(controllers::dashboard::health))
152            .route("/dashboard", get(controllers::dashboard::imp_console))
153            .route("/dashboard/", get(controllers::dashboard::imp_console))
154            .route("/imp-console", get(controllers::dashboard::imp_console))
155            // Display wall — public; the token in the path is the credential
156            // (spec §8.1). The page polls the public display API client-side.
157            .route("/b/:token", get(controllers::dashboard::imp_wall))
158            .route("/dashboard/health", get(controllers::dashboard::health))
159            .route("/dashboard/i18n.js", get(controllers::dashboard::i18n_js))
160            .route("/dashboard/rtl.css", get(controllers::dashboard::rtl_css))
161            .route("/dashboard/locales/en-US.json", get(controllers::dashboard::locale_en_us))
162            .route("/dashboard/locales/ar.json", get(controllers::dashboard::locale_ar))
163            .route("/dashboard/locales/he.json", get(controllers::dashboard::locale_he))
164            .route("/dashboard/locales/fr.json", get(controllers::dashboard::locale_fr))
165            .route("/dashboard/locales/es.json", get(controllers::dashboard::locale_es))
166            .route("/dashboard/locales/de.json", get(controllers::dashboard::locale_de))
167            .route("/dashboard/locales/it.json", get(controllers::dashboard::locale_it))
168            .route("/dashboard/locales/zh.json", get(controllers::dashboard::locale_zh))
169            .route("/dashboard/locales/ko.json", get(controllers::dashboard::locale_ko))
170            .route("/dashboard/locales/ja.json", get(controllers::dashboard::locale_ja))
171            .route("/dashboard/locales/ru.json", get(controllers::dashboard::locale_ru))
172            .route("/dashboard/locales/pt.json", get(controllers::dashboard::locale_pt))
173            .route("/dashboard/locales/hi.json", get(controllers::dashboard::locale_hi))
174            .route("/dashboard/vendor/popper.min.js", get(controllers::dashboard::popper_js))
175            .route("/dashboard/vendor/tippy.min.js", get(controllers::dashboard::tippy_js))
176            .route("/dashboard/vendor/driver.min.js", get(controllers::dashboard::driver_js))
177            // Prong 8 — public first-login wizard (NOT auth-gated; the token in the
178            // path is the credential). Served before the require_auth_for_management layer.
179            //
180            // The two first-login *API* routes (`first-login-status` + `first-login`) are
181            // registered on the `first_login_api` sub-router below, also under the
182            // non-gated `/d/` prefix — NOT here, and NOT under `/api/v1/management/*`
183            // (that prefix is JWT-gated and the wizard holds no token yet).
184            .route("/d/:invite_id/:token", get(controllers::dashboard::first_login));
185
186        // The first-login handlers need `State<AppContext>` but the dashboard router is
187        // `Router<()>`, so we attach state to a nested sub-router. Their absolute `/d/...`
188        // paths land on the public (non-gated) branch of the outer router.
189        let first_login_api = Router::new()
190            .route(
191                "/d/teachers/first-login-status/:invite_id/:token",
192                get(controllers::management::teachers::first_login_status),
193            )
194            .route(
195                "/d/teachers/first-login/:invite_id/:token",
196                post(controllers::management::teachers::first_login),
197            )
198            .with_state(ctx.clone());
199
200        // Gate `/api/v1/management/*` with the JWT middleware, applied as a
201        // `Router::layer` on the outer router (the pattern that works inside Loco's
202        // chain). It is path-scoped: it only enforces JWT for the management prefix and
203        // lets everything else through — so the public dashboard, `/api/v1/health`,
204        // `/api/v1/auth/*`, `/api/v1/telemetry/*`, and the `/d/...` first-login routes
205        // stay accessible without a token.
206        Ok(dashboard
207            .nest("/", first_login_api)
208            .layer(axum::middleware::from_fn(require_auth_for_management))
209            .layer(cors))
210    }
